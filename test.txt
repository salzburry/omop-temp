# Data answers to the open sign-off questions (Q1 unknown T/N, Q2 staging and HPV/p16, Q3 surgery window); Table 1 unchanged
if (!exists("cohort_hn")) source("run_all.R")
options(width = 200)
answers <- list()
answer <- function(sheet, res) { answers[[sheet]] <<- res; message("\n== ", sheet, " =="); print(as.data.frame(res), row.names = FALSE) }
gs_of <- function(x) {                       # GroupStage -> III / IVA / IVB / IVC / IV / I / II / Unknown
  y <- toupper(gsub("(?i)stage|[[:space:]]", "", coalesce(x, ""), perl = TRUE))
  case_when(y %in% c("IVA", "IVB", "IVC", "IV", "III", "II", "I") ~ y, grepl("^III[ABC]$", y) ~ "III", TRUE ~ "Unknown")
}
t_sub <- function(x) {                       # T category keeping T4a / T4b apart
  y <- tn_key(x); full <- "^(Y?[CP]|Y)?(T[0-4])([A-C])?$"
  ifelse(grepl(full, y), paste0(sub(full, "\\2", y), ifelse(sub(full, "\\2", y) == "T4", sub(full, "\\3", y), "")),
         NA_character_)
}
p16_grid <- function(site, hpv)              # AJCC 8th stages these on the p16-positive grid
  norm(hpv) %in% "hpv positive" & (norm(site) %in% "oropharynx" | grepl("unknown primary", norm(site)))

# ---- Q1 Unknown T or N: what GroupStage would imply IF it agreed with T/N (it does not always: see Q1 GroupStage check) ----
q1 <- cohort_hn %>%
  mutate(t_read = tn_status(t_stage, t_cat), n_read = tn_status(n_stage, n_cat), gs = gs_of(groupstage),
         hpv_opc = p16_grid(primarysite, hpvstatus),
         rule = case_when(t_cat %in% c("T1", "T2") & n_cat %in% "N1" ~ "Excluded: T1-T2 N1",
                          (!is.na(t_cat) & !t_cat %in% c("T1", "T2")) | (!is.na(n_cat) & !n_cat %in% "N1") ~
                            "Kept: not T1-T2 N1",
                          TRUE ~ "Undetermined (kept): could be T1-T2 N1, T or N unknown or unrecognised"),
         why = case_when(!grepl("^Undetermined", rule) ~ NA_character_,
                         is.na(t_cat) & is.na(n_cat) ~ "T and N unknown",
                         is.na(t_cat) ~ "T unknown, N1", TRUE ~ "T1-T2, N unknown"),
         implied = case_when(
           is.na(why) ~ NA_character_,
           gs == "Unknown" ~ "GroupStage does not settle it",
           gs %in% c("IV", "IVC") ~ "Stage IV unspecified or IVC (may be M1): any T and N, not settled",
           hpv_opc & gs == "III" ~ "HPV+ oropharynx / unknown primary at III: depends on the staging edition (Q2)",
           gs %in% c("IVA", "IVB") ~ "Stage IVA/IVB: not T1-T2 N1 if GroupStage agrees with T/N",
           !hpv_opc & gs %in% c("I", "II") & why != "T unknown, N1" ~ "Stage I-II: N0 if GroupStage agrees with T/N",
           why == "T1-T2, N unknown" & gs == "III" ~ "Stage III: N1 if GroupStage agrees with T/N (would be excluded)",
           why == "T unknown, N1" & gs == "III" ~ "Stage III: T1-T3, not settled",
           why == "T and N unknown" & gs == "III" ~ "Stage III: T1-T2 N1 or T3 N0-N1, not settled",
           TRUE ~ "GroupStage does not settle it"))
n_unk <- sum(is.na(q1$t_cat) | is.na(q1$n_cat))
answer("Q1 T1-T2 N1 rule", count(q1, rule, name = "patients") %>%
         mutate(note = c(sprintf("If undetermined patients were excluded instead, the primary population would be %d, not %d.",
                                 sum(q1$rule == "Kept: not T1-T2 N1"), sum(q1$rule != "Excluded: T1-T2 N1")),
                         sprintf("All patients with T or N unknown or unrecognised: %d (as on the Attrition sheet); %d of them are kept whatever the unknown value, because their known T or N rules out T1-T2 N1.",
                                 n_unk, n_unk - sum(grepl("^Undetermined", q1$rule))),
                         rep("", max(0, n() - 2)))[seq_len(n())]))
answer("Q1 undetermined", count(filter(q1, !is.na(why)), why, gs, implied, name = "patients"))
answer("Q1 T by N read", count(q1, t_read, n_read, name = "patients"))
q1_chk <- count(filter(q1, rule == "Excluded: T1-T2 N1"), GroupStage = gs, name = "patients with recorded T1-T2 N1")
answer("Q1 GroupStage check", q1_chk)        # how far GroupStage can be trusted to fill in unknown T/N

# ---- Q2 Which AJCC edition do T/N and GroupStage follow; does HPV status come from p16? ----
stage_common <- function(t, n) case_when(    # AJCC 7th (all sites); 8th outside the p16+ grid
  n == "N3" ~ "IVB", t == "T4B" ~ "IVB", t == "T4A" ~ "IVA", t == "T4" ~ "IVA/IVB", n == "N2" ~ "IVA",
  t == "T3" ~ "III", n == "N1" ~ "III", t == "T2" ~ "II", t == "T1" ~ "I", TRUE ~ NA_character_)
stage_p16 <- function(t, n) case_when(       # AJCC 8th clinical, p16-positive grid
  substr(t, 1, 2) == "T4" | n == "N3" ~ "III", t == "T3" | n == "N2" ~ "II",
  t %in% c("T0", "T1", "T2") & n %in% c("N0", "N1") ~ "I", TRUE ~ NA_character_)
fits <- function(expected, gs) !is.na(expected) &
  (expected == gs | (expected == "IVA/IVB" & gs %in% c("IVA", "IVB")) | (gs == "IV" & expected %in% c("IVA", "IVB", "IVA/IVB")))
q2 <- cohort_hn %>%
  mutate(ts = t_sub(t_stage), gs = gs_of(groupstage),
         group = ifelse(p16_grid(primarysite, hpvstatus), "HPV+ oropharynx / unknown primary", "All other patients"),
         exp7 = stage_common(ts, n_cat),
         exp8 = ifelse(group == "All other patients", stage_common(ts, n_cat), stage_p16(ts, n_cat))) %>%
  filter(!is.na(ts), !is.na(n_cat), gs != "Unknown")
answer("Q2 edition check", q2 %>% group_by(group) %>%
         summarise(`patients with T, N and GroupStage` = n(),
                   `fit AJCC 7th` = sum(fits(exp7, gs)), `fit AJCC 8th (clinical)` = sum(fits(exp8, gs)),
                   `fit neither (7th, 8th clinical)` = sum(!fits(exp7, gs) & !fits(exp8, gs)), .groups = "drop") %>%
         mutate(note = ifelse(group != "All other patients",
                              "The editions differ here: fits under each are consistency evidence only, not proof of the edition or of clinical vs pathological staging.",
                              "Both editions use the same grid here: this checks that T/N and GroupStage agree.")))
answer("Q2 misfits", q2 %>% filter(!fits(exp7, gs) & !fits(exp8, gs)) %>%
         count(group, T = ts, N = n_cat, GroupStage = gs, `AJCC 7th` = exp7, `AJCC 8th` = exp8, name = "patients") %>%
         arrange(desc(patients)))
answer("Q2 biomarker tests", src("biomarker") %>% group_by(biomarkername, testtype) %>%   # every test in the table
         summarise(patients = n_distinct(patientid), records = n()) %>% ungroup() %>% collect() %>%
         mutate(across(c(patients, records), as.integer)) %>% arrange(desc(patients)))   # counts arrive as integer64

# ---- Q3 Surgery as first course: surgery from diagnosis to the "start of 1L", five readings (end day included) ----
# Undated records and month/year dates across a window edge stay in; no surgery left = Unresected
ends <- ads %>% select(patientid, dx_date) %>%
  left_join(select(fc_end, patientid, fc_end), by = "patientid") %>%
  left_join(drugs %>% filter(toupper(norm(linesetting)) == "ADVANCED", !is.na(episodedate)) %>%
              group_by(patientid) %>% summarise(adv1 = min(as.Date(episodedate)), .groups = "drop"), by = "patientid") %>%
  left_join(drugs %>% filter(!is.na(episodedate)) %>%
              group_by(patientid) %>% summarise(sys1 = min(as.Date(episodedate)), .groups = "drop"), by = "patientid") %>%
  left_join(src("drugep") %>% filter(patientid %in% !!ids, linenumber == 1, !is.na(linestartdate)) %>%   # the line-start field
              group_by(patientid) %>% summarise(l1 = min(linestartdate, na.rm = TRUE)) %>% collect() %>%
              mutate(l1 = as.Date(l1)), by = "patientid")
rec <- fetch("surgery", surgerytype, sdate = surgerydate, gran = surgerydategranularity) %>%
  mutate(sdate = as.Date(sdate), date_bounds(sdate, gran)) %>% left_join(ends, by = "patientid")
windows <- c("(a) to first ADVANCED-setting drug episode" = "adv1",
             "(b) to end of first course (recurrence / advanced diagnosis)" = "fc_end",
             "(c) to first drug episode of any setting" = "sys1",
             "(d) on or after diagnosis, no end" = NA,
             "(e) to start of line 1 (DrugEpisode LineNumber 1, LineStartDate)" = "l1")
q3 <- list(); q3_new <- list()
for (win in names(windows)) {
  end <- if (is.na(windows[[win]])) as.Date(NA) else rec[[windows[[win]]]]
  pos <- case_when(is.na(rec$sdate) ~ "undated (kept)", is.na(rec$dx_date) ~ "no diagnosis date (kept)",
                   rec$hi < rec$dx_date ~ "before diagnosis",
                   !is.na(end) & rec$lo > end ~ "after the window end",
                   rec$lo < rec$dx_date | (!is.na(end) & rec$hi > end) ~ "month/year date across a window edge (kept)",
                   TRUE ~ "in window")
  q3_new[[win]] <- ads %>% select(patientid, current = surg_group) %>%
    left_join(surg_patient_group(rec[!pos %in% c("before diagnosis", "after the window end"), ]), by = "patientid") %>%
    mutate(current = as.character(current), new = coalesce(surg_group, "Unresected"))
  q3[[win]] <- bind_rows(tibble(definition = win, item = paste("surgery records:", names(table(pos))), n = as.integer(table(pos))),
                       count(q3_new[[win]], current, new) %>% transmute(definition = win, item = paste0("patients: ", current, " -> ", new), n))
}
answer("Q3 surgery windows", bind_rows(q3))
answer("Q3 line 1 settings", src("drugep") %>% filter(patientid %in% !!ids, linenumber == 1) %>%   # what "1L" is here
         distinct(patientid, linesetting) %>% count(linesetting, name = "patients") %>% collect() %>%
         mutate(patients = as.integer(patients)) %>% arrange(desc(patients)))
