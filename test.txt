# T/N reader: T0-T4, Tis / N0-N3 (optional c/p/y prefix, a/b/c dropped); anything else is NA and kept
tn_key <- function(x) toupper(gsub("[[:space:]]", "", coalesce(as.character(x), "")))
tn_cat <- function(x, letter) {
  y <- tn_key(x)
  full <- paste0("^(Y?[CP]|Y)?(", letter, "[0-", if (letter == "T") 4 else 3, "])[A-C]?$")
  out <- ifelse(grepl(full, y), sub(full, "\\2", y), NA_character_)
  if (letter == "T") out[is.na(out) & grepl("^(Y?[CP]|Y)?TIS$", y)] <- "Tis"
  out
}
tn_status <- function(x, cat)
  ifelse(!is.na(cat), cat, ifelse(grepl("^$|^(Y?[CP]|Y)?[TN]X$|^UNKNOWN|NOTDOCUMENTED|NOTREPORTED|NOTAVAILABLE",
                                        tn_key(x)), "Unknown", "Unrecognised"))
site_of <- function(code, sites) {           # first site whose ICD-10 or ICD-9 pattern matches
  k <- toupper(gsub("[[:space:].]", "", code))
  out <- rep(NA_character_, length(k))
  for (i in seq_len(nrow(sites))) out[is.na(out) & grepl(paste0(sites$icd10[i], "|", sites$icd9[i]), k)] <- sites$site[i]
  out
}
date_bounds <- function(d, g) {              # first and last possible day of a day/month/year date
  g <- toupper(coalesce(trimws(g), "DAY"))
  tibble(lo = case_when(g == "YEAR" ~ floor_date(d, "year"), g == "MONTH" ~ floor_date(d, "month"), TRUE ~ d),
         hi = case_when(g == "YEAR" ~ ceiling_date(d, "year", change_on_boundary = TRUE) - 1,
                        g == "MONTH" ~ ceiling_date(d, "month", change_on_boundary = TRUE) - 1, TRUE ~ d))
}

stage_of <- function(x) {
  y <- toupper(gsub("(?i)stage|\\s", "", x, perl = TRUE))
  case_when(grepl("^(0|0IS|T?IS)$", y) ~ "0", grepl("^IV[ABC]?$", y) ~ "IV", grepl("^III[ABC]?$", y) ~ "III",
            grepl("^II[ABC]?$", y) ~ "II", grepl("^I[ABC]?$", y) ~ "I", TRUE ~ "Unknown")
}

table1_long <- function(dat) {
  groups   <- c("Overall", levels(dat$surg_group))
  grp_data <- c(list(Overall = dat), split(dat, dat$surg_group))
  block <- function(var, header) {
    body <- bind_rows(lapply(groups, function(g) {
      d <- grp_data[[g]]
      if (var == "age") {
        q <- quantile(d$age, c(.25, .5, .75), na.rm = TRUE)
        return(tibble(label = c("Mean (sd)", "Median [IQR]"), group = g, n_num = NA, pct = "",
                      n = if (all(is.na(d$age))) "-" else
                        c(sprintf("%.1f (%s)", mean(d$age, na.rm = TRUE),
                                  ifelse(sum(!is.na(d$age)) > 1, sprintf("%.1f", sd(d$age, na.rm = TRUE)), "-")),
                          sprintf("%.1f [%.1f, %.1f]", q[2], q[1], q[3]))))
      }
      x <- if (var == "overall") rep("Overall", nrow(d)) else as.character(d[[var]])
      x[is.na(x)] <- "Unknown"               # an Unknown row only when values are missing
      levs <- if (var == "overall") "Overall" else c(levels(dat[[var]]), if (anyNA(dat[[var]])) "Unknown")
      n <- vapply(levs, function(l) sum(x == l), numeric(1))
      den <- if (var == "overall") nrow(dat) else nrow(d)
      tibble(label = levs, group = g, n_num = n, n = as.character(n),
             pct = if (den > 0) sprintf("%.1f", 100 * n_num / den) else "-")
    }))
    bind_rows(tibble(label = header, group = groups, n = "", n_num = NA, pct = "", is_header = TRUE),
              mutate(body, is_header = FALSE)) %>% mutate(section = header)
  }
  long <- bind_rows(Map(block, t1_vars$var, t1_vars$section)) %>%
    left_join(select(t1_vars, section, baseline), by = "section")
  list(long = long, groups = groups, grp_data = grp_data)
}
wide_table1 <- function(long, groups) long %>%
  pivot_wider(id_cols = c(section, label, is_header, baseline), names_from = group, values_from = c(n, pct)) %>%
  select(section, label, baseline, is_header, all_of(as.vector(rbind(paste0("n_", groups), paste0("pct_", groups)))))

put_numbers <- function(sheet, x, rows, col, fmt) {   # numeric text written as numbers
  k <- which(grepl("^-?[0-9]+(\\.[0-9]+)?$", trimws(x)))
  for (i in k) writeData(wb, sheet, as.numeric(x[i]), startRow = rows[i], startCol = col)
  if (length(k)) addStyle(wb, sheet, createStyle(numFmt = fmt), rows = rows[k], cols = col, stack = TRUE)
}
write_shell <- function(sheet, title, tab, groups, notes) {   # Table 1 / Table 2 in the shell layout
  addWorksheet(wb, sheet, gridLines = FALSE)
  writeData(wb, sheet, title, startRow = 1)
  writeData(wb, sheet, sprintf("Number of cases in Flatiron database, %d to %d", min(dx_years), max(dx_years)),
            startRow = 1, startCol = 2)
  writeData(wb, sheet, "Baseline variable", startRow = 2, startCol = 2)
  for (i in seq_along(groups)) {
    writeData(wb, sheet, groups[i], startRow = 2, startCol = 2 * i + 1)

smd_prop <- function(a, b) {
  d <- sqrt((a * (1 - a) + b * (1 - b)) / 2)
  ifelse(d > 0, (a - b) / d, ifelse(abs(a - b) < 1e-12, 0, sign(a - b) * Inf))
}
smd_multi <- function(a, b) {                # Yang & Dalton 2012, generalised inverse
  if (max(abs(a - b)) < 1e-12) return(0)
  k <- (a + b) > 0; a <- a[k]; b <- b[k]
  x <- a[-1] - b[-1]
  cv <- function(p) { s <- -outer(p, p); diag(s) <- p * (1 - p); s }
  S <- (cv(a[-1]) + cv(b[-1])) / 2
  e <- eigen(S, symmetric = TRUE); keep <- e$values > 1e-12
  Si <- e$vectors[, keep, drop = FALSE] %*% diag(1 / e$values[keep], sum(keep)) %*% t(e$vectors[, keep, drop = FALSE])
  if (!any(keep) || max(abs(x - S %*% Si %*% x)) > 1e-8) return(Inf)   # complete separation
  sqrt(max(0, drop(t(x) %*% Si %*% x)))
}
rate <- function(s) case_when(is.na(s) ~ "Not assessable", abs(s) < 0.1 ~ "Similar",
                              abs(s) < 0.2 ~ "Small difference", TRUE ~ "Meaningful difference")

compare <- function(fi, se, variable, group) {   # one variable, one group, over the categories SEER reports
  d <- left_join(fi, se, by = "label")
  r <- !is.na(d$n) | !is.na(d$pct)
  x <- if (all(!is.na(d$n[r]))) d$n else d$pct
  ok <- sum(r) >= 2 && sum(d$fi_n[r]) > 0 && sum(x[r]) > 0
  d <- d %>% mutate(flatiron_pct = ifelse(r & ok, 100 * fi_n / sum(fi_n[r]), NA),
                    seer_pct = ifelse(r & ok, 100 * x / sum(x[r]), NA),
                    smd = smd_prop(flatiron_pct / 100, seer_pct / 100))
  s <- if (ok) smd_multi(d$flatiron_pct[r] / 100, d$seer_pct[r] / 100) else NA
  i <- which.max(abs(d$flatiron_pct - d$seer_pct))
  list(rows = tibble(Group = group, Variable = variable, Category = d$label, `Flatiron n` = d$fi_n,
                     `Flatiron %` = round(d$flatiron_pct, 1), `SEER %` = round(d$seer_pct, 1),
                     `Difference (pct points)` = round(d$flatiron_pct - d$seer_pct, 1),
                     SMD = round(d$smd, 3), Rating = ifelse(r, rate(d$smd), "No SEER value (left out)")),
       var = tibble(Group = group, Variable = variable, SMD = round(s, 3),
                    Rating = if (ok) rate(s) else "Not assessable (fewer than 2 SEER categories)",
                    `Largest difference` = if (length(i)) d$label[i] else NA,
                    `Flatiron % in categories SEER does not report` = round(100 * sum(d$fi_n[!r]) / max(sum(d$fi_n), 1), 1)))
}

gs_of <- function(x) {                       # GroupStage -> III / IVA / IVB / IVC / IV / I / II / Unknown
  y <- toupper(gsub("(?i)stage|[[:space:]]", "", coalesce(x, ""), perl = TRUE))
  case_when(y %in% c("IVA", "IVB", "IVC", "IV", "III", "II", "I") ~ y, grepl("^III[ABC]$", y) ~ "III", TRUE ~ "Unknown")
}
t_sub <- function(x) {                       # T category keeping T4a / T4b apart
  y <- tn_key(x); full <- "^(Y?[CP]|Y)?(T[0-4])([A-C])?$"
  ifelse(grepl(full, y), paste0(sub(full, "\\2", y), ifelse(sub(full, "\\2", y) == "T4", sub(full, "\\3", y), "")),
         NA_character_)
}

build_mock <- function(n = 400, seed = 20261104) {
  set.seed(seed)
  con <- dbConnect(SQLite(), ":memory:")
  invisible(dbExecute(con, "ATTACH ':memory:' AS clnprw_flatiron_la_hnc_use"))
  put <- function(name, df) dbWriteTable(con, Id(schema = "clnprw_flatiron_la_hnc_use",
                                                table = paste0(name, "_2026m07")), df)
  ids <- sprintf("P%04d", 1:n)
  dx  <- as.Date("2018-06-01") + sample(0:(365 * 5), n, TRUE)
  adv <- ifelse(runif(n) < .3, as.character(dx + sample(200:900, n, TRUE)), NA)

  coh <- data.frame(
    PatientID = ids, DiagnosisDate = as.character(dx), AdvancedDiagnosisDate = adv,
    AdvancedDiagnosisCriteria = ifelse(is.na(adv), NA, "Distant recurrence"),
    GroupStage  = sample(c("III", "IV", "IVA", "IVB", "Unknown/not documented"), n, TRUE),
    PrimarySite = sample(c("Oral cavity", "Oropharynx", "Hypopharynx", "Larynx", "Pharynx NOS",
                           "Tongue NOS", "Unknown primary Head & Neck cancer"), n, TRUE),
    SmokingStatus = sample(c("History of smoking", "No history of smoking"), n, TRUE),
    HPVTested = sample(c("Yes", "No/unknown"), n, TRUE),
    HPVStatus = sample(c("HPV positive", "HPV negative", "HPV equivocal", "Results pending", NA), n, TRUE))
  dxr <- data.frame(PatientID = ids,
    DiagnosisCode = sample(c("C01", "C10.9", "C32.1", "C11.9", "C12", "146.0", "147.1", "C07", "C30.0", "C77.0"),
                           n, TRUE, prob = c(.3, .25, .2, .04, .05, .05, .03, .03, .03, .02))) %>%
    mutate(DiagnosisCodeSystem = ifelse(grepl("^C", DiagnosisCode), "ICD-10-CM", "ICD-9-CM"))
  demo <- data.frame(PatientID = ids,
    BirthSex = sample(c("M", "F", NA), n, TRUE, prob = c(.74, .25, .01)),
    BirthYear = sample(c(1935:2004, NA), n, TRUE),
    Race = sample(c("White", "Black or African American", "Asian", "Other Race",
                    "Hispanic or Latino", NA), n, TRUE, prob = c(.6, .1, .05, .1, .05, .1)),
    Ethnicity = sample(c("Not Hispanic or Latino", "Hispanic or Latino", NA), n, TRUE))
  prac <- data.frame(PatientID = sample(ids, n * 1.3, TRUE),
    PracticeType = sample(c("COMMUNITY", "ACADEMIC"), n * 1.3, TRUE, prob = c(.8, .2))) %>%
    mutate(PracticeID = sprintf("%s%d", substr(PracticeType, 1, 2), seq_along(PatientID) %% 5 + 1))

  s_ids <- sample(ids, n * .8)
  s_types <- c("Primary tumor resection only", "Primary tumor resection and lymph node dissection",
               "Lymph node dissection only", "Unknown/not specified", "Endoscopic mucosal resection")
  surg <- data.frame(PatientID = c(s_ids, sample(s_ids, 40)),
    SurgeryType = sample(s_types, length(s_ids) + 40, TRUE, prob = c(.3, .3, .2, .15, .05)))
  surg$SurgeryDate <- as.character(dx[match(surg$PatientID, ids)] + 30)

  d_ids <- sample(ids, n * .7); nd <- length(d_ids)
  drugs <- data.frame(PatientID = d_ids,
    LineSetting = sample(c("CONCURRENT", "ADJUVANT", "INDUCTION CHEMO", "NEOADJUVANT", "EARLY",
                           "RADIATION THERAPY", "ADVANCED", "LOCOREGIONAL"), nd, TRUE),
    DrugName = sample(c("cisplatin", "carboplatin", "cetuximab", "pembrolizumab", "ondansetron"),
                      nd, TRUE),
    EpisodeDate = as.character(dx[match(d_ids, ids)] + 60))

  r_ids <- sample(ids, n * .7)
  rt <- data.frame(PatientID = r_ids,
    RTStartDate = as.character(dx[match(r_ids, ids)] + sample(-60:400, length(r_ids), TRUE)),
    RTStartDateGranularity = sample(c("Day", "Month", "Year"), length(r_ids), TRUE, prob = c(.9, .05, .05)),
    RTSite = sample(c("Primary site (Includes regional lymph node involvement)",
                      "Distant lymph nodes", "Distant site", NA), length(r_ids), TRUE,
                    prob = c(.7, .1, .1, .1)))
  lr_ids <- sample(ids, n * .1)
  lrec <- data.frame(PatientID = lr_ids, LocalRecurSite = "Local site",
                     LocalRecurDate = as.character(dx[match(lr_ids, ids)] + sample(150:700, length(lr_ids), TRUE)))
  m_ids <- sample(ids, n * .05)
  mrec <- data.frame(PatientID = m_ids,
                     DistantMetDate = as.character(dx[match(m_ids, ids)] + sample(150:700, length(m_ids), TRUE)))
  prac <- distinct(prac)                                  # 0-3 visits per patient-practice, -200 to +400 days
  visit <- prac[rep(seq_len(nrow(prac)), sample(0:3, nrow(prac), TRUE)), c("PatientID", "PracticeID")]
  visit$VisitDate <- as.character(dx[match(visit$PatientID, ids)] + sample(-200:400, nrow(visit), TRUE))
  coh$TStage <- sample(c("T1", "T2", "T3", "T4a", "T4b", "TX", "Unknown/not documented", NA), n, TRUE)
  coh$NStage <- sample(c("N0", "N1", "N2a", "N2b", "N2c", "N3", "NX", NA), n, TRUE)
  coh$NStage[coh$TStage %in% c("T1", "T2") & coh$NStage %in% "N1"] <- "N2a"   # random patients never T1-T2 N1

  # ---- fixtures ----
  for (f in fixtures) {
    coh <- rbind(coh, data.frame(PatientID = f$id, DiagnosisDate = f$dx, AdvancedDiagnosisDate = f$adv,
      AdvancedDiagnosisCriteria = NA, GroupStage = f$stage, PrimarySite = f$site,
      SmokingStatus = "History of smoking", HPVTested = "Yes", HPVStatus = f$hpv, TStage = f$t, NStage = f$n))
    for (dc in f$dx_codes) dxr <- rbind(dxr, data.frame(PatientID = f$id, DiagnosisCode = dc[1],
                                                       DiagnosisCodeSystem = dc[2]))
    demo <- rbind(demo, data.frame(PatientID = f$id, BirthSex = f$birthsex, BirthYear = f$birthyear,
                                   Race = f$race, Ethnicity = f$eth))
    for (v in f$visits) {
      prac <- rbind(prac, data.frame(PatientID = f$id, PracticeType = v[1], PracticeID = paste0("FX_", v[1])))
      if (!is.na(v[2])) visit <- rbind(visit, data.frame(PatientID = f$id, PracticeID = paste0("FX_", v[1]),
                                                         VisitDate = as.character(as.Date(f$dx) + as.numeric(v[2]))))
    }
    for (s in f$surg) surg <- rbind(surg, data.frame(PatientID = f$id, SurgeryType = s,
                                                     SurgeryDate = as.character(as.Date(f$dx) + f$surg_day)))
    for (d in f$drugs) drugs <- rbind(drugs, data.frame(PatientID = f$id, LineSetting = d[1], DrugName = d[2],
      EpisodeDate = as.character(as.Date(f$dx) + if (length(d) > 2) as.numeric(d[3]) else 0)))   # optional day offset
    for (r in f$rt) rt <- rbind(rt, data.frame(PatientID = f$id, RTStartDate = r[1],
                                               RTStartDateGranularity = r[2], RTSite = r[3]))
    if (!is.na(f$localrec)) lrec <- rbind(lrec, data.frame(PatientID = f$id, LocalRecurSite = "Local site",
                                                           LocalRecurDate = f$localrec))
    if (!is.na(f$metrec)) mrec <- rbind(mrec, data.frame(PatientID = f$id, DistantMetDate = f$metrec))
  }
  drugs <- drugs %>% mutate(DrugCategory = drug_cat(DrugName),
                            DetailedDrugCategory = unname(drug_class[DrugName]),
                            EnhancedCohort = "LAHNC")

  put("t_enhanced_lahnc", select(coh, -TStage, -NStage))      # T and N are in DeepDiagnosis, as delivered
  put("t_enhanced_lahnc_deep_diagnosis", transmute(coh, PatientID, TStage, NStage, MStage = "M0", DiseaseStatus = NA_character_))
  put("t_diagnosis", dxr)
  put("t_demographics", demo)
  put("t_practice", distinct(prac))
  put("t_visit", mutate(visit, VisitType = "Treatment"))
  put("t_enhanced_lahnc_primary_surgery", mutate(surg, SurgeryDateGranularity = "Day"))
  put("t_drug_episode", drugs)
  put("t_enhanced_lahnc_radiation_therapy", rt)
  put("t_enhanced_lahnc_localrecurrence", lrec)
  put("t_enhanced_lahnc_metrecurrence", mrec)
  put("t_enhanced_lahnc_biomarker", data.frame(PatientID = c("P0001", "P0002", "P0003"), BiomarkerName = c("HPV", "HPV", "PD-L1"),
    TestType = c("IHC", "PCR", "IHC"), Assay = c("p16 IHC", NA, "22C3"), BiomarkerStatus = c("Positive", "Negative", "Positive")))
  con
}

run <- function(con, overrides = list(), steps = "01", seer = "inputs/seer_table2_laHNSCC.csv") {
  s0 <- sub("^con <- connect\\(\\)$", "con <- mock_con", readLines("R/00_setup.R"))
  stopifnot(sum(s0 == "con <- mock_con") == 1)
  assign("mock_con", con, envir = globalenv())
  invisible(capture.output(suppressMessages({
    eval(parse(text = s0), envir = globalenv())
    assign("out_dir", file.path(mock_dir, "out"), envir = globalenv())
    assign("out_stem", "MOCK", envir = globalenv())
    assign("seer_file", seer, envir = globalenv())
    for (nm in names(overrides)) assign(nm, overrides[[nm]], envir = globalenv())
    for (st in steps) sys.source(list.files("R", paste0("^", st, "_"), full.names = TRUE), envir = globalenv())
  })))
}
