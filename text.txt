MC_CATALOG <- "hive_metastore"
MC_SCHEMA  <- "clnprw_optum_mc_full_use"
MC_CUT     <- "2025q4"                 # tables are t_<name>_<CUT>
MIN_CELL   <- 11L                      # small-cell suppression

# Open a connection — the established Domino/Databricks ODBC pattern:
# a configured DSN (default "RWDE"), token supplied as `pwd` via env var, an explicit
# timeout, and bigint = "numeric" so SQL COUNT/aggregate results don't come back as
# integer64 (which breaks arithmetic/masking downstream). Never hard-code the token.
mc_connect <- function(dsn     = Sys.getenv("DATABRICKS_DSN", "RWDE"),
                       pwd     = Sys.getenv("DATABRICKS_PWD", ""),
                       timeout = 120,
                       bigint  = "numeric",
                       ...) {
  if (!nzchar(pwd)) stop("DATABRICKS_PWD is not set (Domino env var / project secret).")
  DBI::dbConnect(odbc::odbc(), dsn = dsn, pwd = pwd, timeout = timeout, bigint = bigint, ...)
}

# Lazy handle to a deployed table: mc_tbl(con, "diagnosis") -> t_diagnosis_2025q4
# rename_with(tolower): Databricks reports columns UPPERCASE and dbplyr's R-side name
# check is case-sensitive, so we lowercase them here to match the lowercase field
# references used throughout the pipeline (else every filter errors "object not found").
mc_tbl <- function(con, name) {
  dplyr::tbl(con, dbplyr::in_catalog(MC_CATALOG, MC_SCHEMA, paste0("t_", name, "_", MC_CUT))) |>
    dplyr::rename_with(tolower)
}

# Small-cell masking for display (true 0 kept; 1..MIN_CELL-1 -> "<MIN_CELL"; NA -> "n/a")
mc_mask <- function(n) {
  dplyr::case_when(
    is.na(n)                 ~ "n/a",
    n > 0 & n < MIN_CELL     ~ paste0("<", MIN_CELL),
    TRUE                     ~ as.character(as.integer(n))
  )
}

# Count distinct ptid in a lazy tbl (materialise-free scalar pull)
n_ptid <- function(tb) {
  tb |> dplyr::summarise(n = dplyr::n_distinct(ptid)) |> dplyr::pull(n) |> as.integer()
}

# ── cohort entry: >=2 confirmed colon (C18) dx dates; first_dx = earliest ───
rdq1_cohort_dx <- function(src) {
  src$dx |>
    filter(str_trim(tolower(diagnosis_status)) == "diagnosis of",              # value-casing robust (matches rdq1_full.R)
           str_detect(upper(diagnosis_cd_type), "^ICD"),                       # ICD-9/10 only (lib parity)
           str_detect(upper(diagnosis_cd), RDQ1_COLON_ICD_REGEX)) |>    # upper() -> robust to case
    mutate(dxd = to_date(diag_date)) |>
    group_by(ptid) |>
    summarise(first_dx = min(dxd, na.rm = TRUE),
              n_dx     = n_distinct(dxd), .groups = "drop") |>
    filter(n_dx >= 2) |>
    select(ptid, first_dx)
}

# ── STEP 1: colon-cancer surgery (the surgery code list) ────────────────────
# Match on the NORMALIZED proc_cd VALUE (upper + strip non-alnum) against the authoritative surgery
# code list. The patient-side proc_cd_type is NOT required — its value spellings are UNCONFIRMED
# ('CPT'/'HCPCS'/'ICD10PCS' vs 'CPT4'/'ICD-10-PCS'/NULL) — so requiring a specific one could silently
# drop rows. ⚠️ FLAGS: (1) the `procedure` table is documented
# as CPT/HCPCS; ICD-10-PCS may not be captured there, so the `0DT*` PCS arm may match
# nothing (confirm). (2) proc_cd_type is available for a STRICTER match once its spelling is confirmed.
rdq1_step1_surgery <- function(src, fdx, include_nonop_hcpcs = TRUE,
                               study_start = RDQ1_STUDY_START, study_end = RDQ1_STUDY_END) {
  surg_codes <- if (isTRUE(include_nonop_hcpcs)) c(RDQ1_SURG_CPT, RDQ1_SURG_HCPCS_NONOP) else RDQ1_SURG_CPT
  out <- src$proc |>
    mutate(px  = regexp_replace(upper(proc_code), "[^0-9A-Z]", ""),   # deployed col: proc_code (not proc_cd)
           pdt = to_date(proc_date)) |>
    filter(px %in% surg_codes | str_detect(px, RDQ1_SURG_PCS_REGEX)) |>
    semi_join(fdx, by = "ptid") |>
    transmute(ptid, surgery_date = pdt) |>
    distinct()
  # STUDY-PERIOD anchor on the surgery/index date (documented window; togglable via study_start/end;
  # NULL disables). Pre-2020 lookback stays usable — only the selected index date is constrained.
  if (!is.null(study_start)) out <- out |> filter(surgery_date >= to_date(study_start))
  if (!is.null(study_end))   out <- out |> filter(surgery_date <= to_date(study_end))
  out
}

# ── STEP 2: earliest surgery whose [-183d, 0] window contains first_dx ───────
# first_dx is the GLOBAL-min confirmed colon dx, so "first_dx within [sdt-183, sdt]" already
# enforces the pre-window washout (no earlier colon dx).
rdq1_step2_newdx <- function(surg, fdx) {
  surg |>
    inner_join(fdx, by = "ptid") |>
    filter(datediff(surgery_date, first_dx) >= 0,
           datediff(surgery_date, first_dx) <= 183) |>
    group_by(ptid, first_dx) |>
    summarise(surgery_date = min(surgery_date), .groups = "drop")
}

# ── STEP 3: stage T1-4 N1-2 or T4N0 (EO TNM; N-missing -> positive nodes) ────
# EO-only -> populates for use_onc cohorts. Peri-op window [first_dx-30, surgery+90].
rdq1_step3_stage <- function(con, base) {
  # colon concept keys. Stage-window anchor = the NEOPLASM-histology diagnosis date. "Stage at
  # diagnosis" links via onc_neoplasm_histology `neoplasm_dx_date`, and if that is missing the
  # neoplasm `note_date` near structured first-dx — so the anchor is coalesce(neoplasm_dx_date,
  # note_date), NOT the onc_tnm row's own note_date.
  keys <- mc_tbl(con, "onc_neoplasm_histology") |>
    semi_join(base, by = "ptid") |>
    filter(str_trim(tolower(neoplasm_qualifier)) == "actual",
           upper(str_trim(neoplasm_type)) %in% RDQ1_COLON_NEO) |>
    transmute(ptid, neoplasm_histology_key,
              neo_anchor_date = coalesce(to_date(neoplasm_dx_date), to_date(note_date))) |>
    inner_join(base, by = "ptid")                      # brings surgery_date, first_dx

  # positive-node fallback (stand-alone onc_lymph_node), peri-op window.
  # ⚠️ DATA-DICT: lymph_numeric is "# positive/NEGATIVE nodes OR nodes biopsied" — ambiguous
  # ALONE; lymph_narrative (documented value 'positive lymph node') disambiguates. So require a positive
  # narrative AND count > 0.
  posnodes <- mc_tbl(con, "onc_lymph_node") |>
    inner_join(base, by = "ptid") |>
    mutate(nd = to_date(note_date)) |>
    filter(str_trim(tolower(lymph_narrative)) == "positive lymph node",
           lymph_numeric > 0,
           datediff(nd, first_dx) >= -30,
           datediff(nd, surgery_date) <= 90) |>
    distinct(ptid) |>
    mutate(has_posnode = 1L)

  # select only the TNM columns we need (avoids onc_tnm.ptid / .note_date colliding with keys)
  # CONFIRMED deployed columns: onc_tnm has `t`, `n` (grade) and the FK `neoplasm_histology_key`
  # (also t_prefix/t_suffix, n_prefix/n_suffix, m/m_prefix/m_suffix, stage_prefix). Verified via DESCRIBE.
  mc_tbl(con, "onc_tnm") |>
    select(neoplasm_histology_key, t, n) |>
    inner_join(keys, by = "neoplasm_histology_key") |>
    mutate(tnum = sql("try_cast(regexp_extract(t,'[1-4]',0) as int)"),
           nnum = sql("try_cast(regexp_extract(n,'[0-9]',0) as int)"),
           nraw = sql("regexp_extract(n,'[0-9]',0)")) |>
    left_join(posnodes, by = "ptid") |>
    mutate(has_posnode = coalesce(has_posnode, 0L)) |>
    filter(tnum >= 1, tnum <= 4,
           (nnum > 0) |
             (tnum == 4 & nnum == 0) |
             ((nraw == "" | is.na(nnum)) & has_posnode == 1L),
           datediff(neo_anchor_date, first_dx) >= -30,       # stage window on coalesce(neoplasm_dx_date,note_date)
           datediff(neo_anchor_date, surgery_date) <= 90) |>
    distinct(ptid, surgery_date, first_dx)
}

# ── STEP 4: no metastatic dx prior to surgery (RDQ-1: C77+C78+C79) ───────────
# Table 1a step 4 is tagged [from EHR] -> EHR-coded ICD ONLY (criterion 4 = C77/C78/C79 when
# include_c77). The EO onc_metastatic_location overlay is a Table 1b / curated-sensitivity signal,
# NOT part of the [from EHR] step — deliberately EXCLUDED here.
rdq1_step4_nomets <- function(con, src, step3, include_c77 = TRUE) {
  mets <- src$dx |>
    filter(str_trim(tolower(diagnosis_status)) == "diagnosis of",              # value-casing robust (matches rdq1_full.R)
           str_detect(upper(diagnosis_cd_type), "^ICD"),
           str_detect(upper(diagnosis_cd), rdq1_mets_regex(include_c77))) |>
    mutate(mdt = to_date(diag_date)) |>
    group_by(ptid) |> summarise(mdt = min(mdt), .groups = "drop")
  out <- step3 |> left_join(mets, by = "ptid") |>
    filter(is.na(mdt) | datediff(mdt, surgery_date) >= 0)   # keep if no mets before surgery
  list(step4 = out |> select(ptid, surgery_date, first_dx), mets = mets)
}

# ── STEP 5: dMMR+ or MSI-H ───────────────────────────────────────────────────
# Single-primary restriction for the STAND-ALONE biomarker table (onc_biomarker has no
# neoplasm_histology_key; links by PTID + note_date). Excludes patients with an 'actual'
# neoplasm OUTSIDE the colon set. ⚠️ This is a SOUND INFERENCE (attribute a
# biomarker only when no other primary/stand-alone marker is present) — NOT an explicit rule.
rdq1_multiprimary <- function(con, cohort_tbl) {
  mc_tbl(con, "onc_neoplasm_histology") |>
    semi_join(cohort_tbl, by = "ptid") |>
    filter(str_trim(tolower(neoplasm_qualifier)) == "actual",
           !(upper(str_trim(neoplasm_type)) %in% RDQ1_COLON_NEO)) |>
    distinct(ptid)
}

rdq1_step5_dmmr <- function(con, step4) {
  # DOCUMENTED dMMR/MSI-H rule: "a MUTATION in any of MLH1, MSH2, MSH6, PMS2 -> dMMR/MSI-H".
  # So the 4 HGNC genes ONLY — generic 'MMR' is NOT a stored biomarker value (removed).
  # ⚠️ The RESULT-value strings below are OUR interpretation, NOT documented: narrative_result values live
  # in the ONC_NARRATIVE_RESULT_LU lookup (not enumerated here), so 'high|unstable|positiv' (MSI-H) and
  # 'loss|absent|deficien' + gene_characteristics 'mutat|loss|deletion' (mutation) are BEST-EFFORT proxies
  # for the "a mutation" rule — CONFIRM against ONC_NARRATIVE_RESULT_LU on the instance. (An MSI biomarker
  # row is likewise not enumerated.) biomarker/gene_characteristics/temporal_status ARE documented.
  mp <- rdq1_multiprimary(con, step4)                              # single-primary restriction (inferred)
  dmmr <- mc_tbl(con, "onc_biomarker") |>
    semi_join(step4, by = "ptid") |>
    anti_join(mp, by = "ptid") |>
    # exclude PLANNED (not-yet-performed) tests. Documented status historical/contemporary/planned; stored
    # code likely 'plan' (Character(7)) — exclude both spellings.
    filter(!(str_trim(lower(coalesce(biomarker_temporal_status, ""))) %in% c("plan","planned"))) |>
    filter(
      (str_detect(upper(biomarker), "MSI") &
         (str_detect(lower(coalesce(narrative_result, "")), "high|unstable|positiv") |
            str_detect(upper(biomarker), "MSI-H"))) |
      (str_detect(upper(biomarker), "MLH1|MSH2|MSH6|PMS2") &                # documented dMMR genes only
         (str_detect(lower(coalesce(narrative_result, "")), "loss|lost|absent|deficien|mutat") |
            str_detect(lower(coalesce(gene_characteristics, "")), "mutat|loss|deletion")))
    ) |>
    transmute(ptid, bio_dt = to_date(note_date))
  step4 |> inner_join(dmmr, by = "ptid") |>
    filter(datediff(bio_dt, first_dx) >= 0) |>                      # biomarker on/after first dx
    distinct(ptid, surgery_date, first_dx)
}

# ── STEP 6: IO use after dx, before any metastatic dx (11-agent list) ────────
rdq1_step6_io <- function(src, step5, mets) {
  io <- src$med |>
    semi_join(step5, by = "ptid") |>
    filter(upper(str_trim(generic_desc)) %in% RDQ1_IO,      # str_trim to normalize whitespace
           str_trim(lower(action)) %in% RDQ1_IO_ACTIONS) |>
    transmute(ptid, io_dt = to_date(action_date))
  io |>
    inner_join(step5, by = "ptid") |>
    left_join(mets, by = "ptid") |>
    filter(datediff(io_dt, first_dx) > 0,                    # io after first dx
           is.na(mdt) | datediff(mdt, io_dt) > 0) |>          # io before any mets
    distinct(ptid)
}

# ── one column: attrition Ns for a single {cohort, table_variant} ───────────
rdq1_attrition_column <- function(con, cohort,
                                  table_variant = "1a",
                                  include_c77 = TRUE,
                                  include_nonop_hcpcs = TRUE,
                                  study_start = RDQ1_STUDY_START,
                                  study_end = RDQ1_STUDY_END) {
  steps <- c("1_surgery","2_newdx","3_stage","4_nomets","5_dmmr","6_io")
  na_col <- tibble::tibble(step = steps, n = NA_integer_)

  # R sources ONLY cohort 1 today (rdq1_sources ignores `cohort`). Every other label — 2/2a/2b (+Optum
  # claims), 3/3a/3b (third-party) and 4 (Market Clarity IHD) — needs claims/IHD source unions NOT wired in
  # R, so return n/a rather than emit cohort-1 EHR data under a different label. WHITELIST (not a 2/3/4
  # blacklist): a label such as '2a' must NOT slip through and be computed+mislabeled. So R produces
  # cohort 1 only; everything else = n/a.
  if (cohort != "1") return(na_col)
  if (table_variant != "1a")                           # 1b (all-EO) re-sourcing is the open delta
    stop("table_variant '1b' (all-EO re-sourcing) is not implemented yet.")

  use_onc <- TRUE                                       # cohort 1 always has EO (only cohort reachable here)
  src <- rdq1_sources(con, cohort, use_onc)
  fdx <- rdq1_cohort_dx(src) |> compute()

  s1 <- rdq1_step1_surgery(src, fdx, include_nonop_hcpcs, study_start, study_end) |> compute()
  s2 <- rdq1_step2_newdx(s1, fdx) |> compute()
  n <- c(`1_surgery` = n_ptid(s1), `2_newdx` = n_ptid(s2))

  if (use_onc) {
    s3  <- rdq1_step3_stage(con, s2) |> compute()
    s4l <- rdq1_step4_nomets(con, src, s3, include_c77)   # ICD-only mets (Table 1a [from EHR])
    s4  <- s4l$step4 |> compute()
    s5  <- rdq1_step5_dmmr(con, s4) |> compute()
    s6  <- rdq1_step6_io(src, s5, s4l$mets) |> compute()
    n <- c(n, `3_stage` = n_ptid(s3), `4_nomets` = n_ptid(s4),
           `5_dmmr` = n_ptid(s5), `6_io` = n_ptid(s6))
  } else {
    n <- c(n, `3_stage` = 0L, `4_nomets` = 0L, `5_dmmr` = 0L, `6_io` = 0L)   # cohort 4 (IHD): no EO
  }
  tibble::tibble(step = steps, n = as.integer(n[steps]))
}

# tb is an already-built lazy dbplyr table (or NULL if its construction failed upstream);
# translation errors surface at render/collect and are reported here, never aborting the run.
probe <- function(label, tb) {
  cat("\n===", label, "===\n")

#     (clm_med_diagnosis.diag/icd_flag, clm_med_services.proc_cd/ndc, clm_rx.gnrc_nm/ndc,
#     dates fst_dt/fill_dt) - a separate claims implementation (IO billed as J-codes/NDC,
#     no 'diagnosis_status'). Not included here to avoid shipping unverified claims logic.
#
#   Table 1a = coded EHR + EO (steps 1/2/4/6 coded; 3/5 from EO).
#   Table 1b = all-from-EO where EO holds the concept (dx from neoplasm_dx_date, mets from
#              onc_metastatic_location KEYED to the colon primary, IO from onc_medications EHR+NLP
#              arm); surgery stays structured Procedure (EO has no surgery event - flagged).
#
# CAVEATS - what "verified" means and does NOT mean:
#   * VERIFIED = every COLUMN name checked against the deployed schema + syntax balanced.
#   * Filter VALUES are now CONFIRMED FROM THE OPTUM DATA DICT (see R/README.md for page cites):
#     diagnosis_status = "Diagnosis of" (MC dict p5, full domain listed); diagnosis_cd_type in ICD9/ICD10 (p5);
#     neoplasm_qualifier = "actual" (EO p16, shown as both actual/Actual - hence the tolower here);
#     neoplasm_type = organ/tissue, colon->"colon" (EO p11-12); data_source EHR arm = proc/rx_presc/rx_adm
#     (EO p21 source-table list); action in started/stopped/is_on/suspended/administered/filled (EO p20);
#     dMMR/MSI-H = mutation in MLH1/MSH2/MSH6/PMS2 (EO p11, VERBATIM); mets_qualifier in {actual, absent}
#     (onco deck p25 - filter to "actual" excludes negated mets); TNM t/n format 1a/4d/3c/x (deck p22).
#     STILL to confirm live (dict didn't enumerate): exact colon token (upper()-safe), data_source NLP value
#     (1b only), and the supplementary MSI-H narrative_result strings.
#   * Step 5 (dMMR/MSI-H) is EO-BIOMARKER-ONLY - the structured-Lab (LOINC) arm is NOT wired, so step 5
#     may UNDERCOUNT until that value-set is added (biomarker data live in BOTH EO Biomarker and Labs).
#   * Step 6 (IO) matches onc_medications.generic_desc ONLY. onc_medications has NO NDC / HCPCS / J-code
#     column, so the RDQ's NDC+J-code IO matching applies to the CLAIMS path (cohorts 2/4), not here;
#     cohort-1 step 6 may undercount a row whose generic_desc is blank.

# EO stage (step 3) - takes a `base` (ptid, surgery_date, first_dx); returns qualifying ptids
eo_stage <- function(base) {
  keys <- Tab("onc_neoplasm_histology") |> semi_join(base, by = "ptid") |>
    filter(str_trim(tolower(neoplasm_qualifier)) == "actual", upper(str_trim(neoplasm_type)) %in% COLON_NEO) |>
    transmute(ptid, neoplasm_histology_key,
              anchor = coalesce(to_date(neoplasm_dx_date), to_date(note_date))) |>
    inner_join(base, by = "ptid")
  posnodes <- Tab("onc_lymph_node") |> inner_join(base, by = "ptid") |>
    mutate(nd = to_date(note_date)) |>
    filter(str_trim(tolower(lymph_narrative)) == "positive lymph node", lymph_numeric > 0,
           datediff(nd, first_dx) >= -30, datediff(nd, surgery_date) <= 90) |>
    distinct(ptid) |> mutate(has_pos = 1L)
  Tab("onc_tnm") |> select(neoplasm_histology_key, t, n) |>
    inner_join(keys, by = "neoplasm_histology_key") |>
    mutate(tnum = sql("try_cast(regexp_extract(t,'[1-4]',0) as int)"),
           nnum = sql("try_cast(regexp_extract(n,'[0-9]',0) as int)"),
           nraw = sql("regexp_extract(n,'[0-9]',0)")) |>
    left_join(posnodes, by = "ptid") |> mutate(has_pos = coalesce(has_pos, 0L)) |>
    filter(tnum >= 1, tnum <= 4,
           (nnum > 0) | (tnum == 4 & nnum == 0) | ((nraw == "" | is.na(nnum)) & has_pos == 1L),
           datediff(anchor, first_dx) >= -30, datediff(anchor, surgery_date) <= 90) |>
    distinct(ptid, surgery_date, first_dx)
}

# EO dMMR / MSI-H (step 5) - takes step4 (ptid, surgery_date, first_dx)
eo_dmmr <- function(step4) {
  mp <- Tab("onc_neoplasm_histology") |> semi_join(step4, by = "ptid") |>
    filter(str_trim(tolower(neoplasm_qualifier)) == "actual", !(upper(str_trim(neoplasm_type)) %in% COLON_NEO)) |>
    distinct(ptid)
  dmmr <- Tab("onc_biomarker") |> semi_join(step4, by = "ptid") |> anti_join(mp, by = "ptid") |>
    filter(!(str_trim(lower(coalesce(biomarker_temporal_status, ""))) %in% c("plan","planned"))) |>
    filter(
      (str_detect(upper(biomarker), "MSI") &
         (str_detect(lower(coalesce(narrative_result, "")), "high|unstable|positiv") |
            str_detect(upper(biomarker), "MSI-H"))) |
      (str_detect(upper(biomarker), "MLH1|MSH2|MSH6|PMS2") &
         (str_detect(lower(coalesce(narrative_result, "")), "loss|lost|absent|deficien|mutat") |
            str_detect(lower(coalesce(gene_characteristics, "")), "mutat|loss|deletion")))
    ) |> transmute(ptid, bio_dt = to_date(note_date))
  step4 |> inner_join(dmmr, by = "ptid") |> filter(datediff(bio_dt, first_dx) >= 0) |>
    distinct(ptid, surgery_date, first_dx)
}

# surgery (step 1) - structured Procedure; same for 1a and 1b (EO has no surgery event)
surgery_step1 <- function(fdx) {
  s <- Tab("procedure") |>
    mutate(px = regexp_replace(upper(proc_code), "[^0-9A-Z]", ""), pdt = to_date(proc_date)) |>
    filter(px %in% SURG_CODES | str_detect(px, SURG_PCS)) |>
    semi_join(fdx, by = "ptid") |> transmute(ptid, surgery_date = pdt) |> distinct()
  if (!is.na(STUDY_START)) s <- s |> filter(surgery_date >= to_date(STUDY_START))
  if (!is.na(STUDY_END))   s <- s |> filter(surgery_date <= to_date(STUDY_END))
  compute(s)
}

# IO step 6 - `provider_notes_ok` = TRUE for Table 1b (adds the NLP provider_notes source), FALSE for 1a.
# ALWAYS restrict data_source to the EHR/NLP arm - NEVER leave onc_medications unfiltered, or claims-med
# rows (clm_rx / clm_med_serv data_source values) would leak into cohort 1, which is a no-claims cohort.
io_step6 <- function(step5, mets, provider_notes_ok) {
  srcs <- if (provider_notes_ok) c(MED_SRC, "provider_notes") else MED_SRC
  Tab("onc_medications") |>
    filter(str_trim(tolower(data_source)) %in% srcs) |>
    semi_join(step5, by = "ptid") |>
    filter(upper(str_trim(generic_desc)) %in% IO, str_trim(lower(action)) %in% IO_ACT) |>
    transmute(ptid, io_dt = to_date(action_date)) |>
    inner_join(step5, by = "ptid") |> left_join(mets, by = "ptid") |>
    filter(datediff(io_dt, first_dx) > 0, is.na(mdt) | datediff(mdt, io_dt) > 0) |>
    distinct(ptid)
}

## 4. Non-standard choices & embedded assumptions in RDQ-1 *itself*

Separately from "does it match our pipeline" (§5), several things RDQ-1 specifies are **not standard
RWD/epidemiology practice**, or are **assumptions the request leaves unstated**. These are inherent to the
request as written and should be surfaced to the requestor regardless of how we implement — each is graded
by where it could **bias the cohort**. Everything below reflects the request as written and the Optum data
realities documented in `docs/data-model-reference.md`.

### Internally inconsistent / clinically questionable
- **(A) C77.x in the metastatic exclusion can conflict with an N+ cohort.** C77 = *secondary malignant
  neoplasm of lymph nodes*; it **can overlap with the regional nodal (N+) disease the cohort is trying to
  retain** (criterion 3 deliberately selects **N1–2**). It is not the case that every C77 code is regional —
  but excluding C77 wholesale risks dropping target patients. *(→ decision in §5.1.)*
- **(D) Lymph-node count used as an N-stage proxy.** "In absence of N staging, no. of lymph nodes is >0"
  substitutes a node *count* for N stage. The `onc_lymph_node` table is **stand-alone (no
  `neoplasm_histology_key`)** and carries `lymph_numeric` (a node **count** that may be
  positive/negative/**or** biopsied nodes — **not** unambiguously "# positive") and
  `lymph_numeric_scale` (# **examined**) — conflating "nodes >0" with "positive nodes >0" would misclassify
  node-negative resections as N+. Recall-limited and attribution-ambiguous for multi-primary patients.
- **(J) The surgery list includes non-surgical HCPCS.** Per the code list's **own descriptions**, these
  include screening colonoscopy (G0105/G0120/G0121), disease-status/demonstration codes (G9084–G9089,
  G9659/G9660…), and history-of-colectomy / CRC-registry codes (G9711, M1295). These are **not operative
  events** and would inflate step 1. *(The code descriptions come from the RDQ code list itself; whether to
  keep them is a **clinical/coding-review** decision — see §5.4.)*
- **(I) The IO list is broad for the indication** *(clinical-review concern, not a document fact)*. The
  list has 11 checkpoint agents including **ipilimumab (CTLA-4)** and several agents that are not part
  of standard CRC regimens (avelumab, toripalimab, tislelizumab, cosibelimab). This is a clinical judgement,
  flagged for confirmation: "any IO" ≠ "IO given for this colon cancer," especially with the
  open-ended window in (H). RDQ-1 lists them, so the default is to include all 11.

### Blind to known NLP under-capture (false exclusions)
- **(E) Treats stage as a clean filter, but it is enrichment-derived and recall-limited.** Step 3 is
  **not** `[from EHR]`-tagged in either table — stage comes from **EO TNM / lymph-node** capture in **both**
  Table 1a *and* 1b. So the constraint is not "structured EHR staging is absent"; it is that **EO TNM
  extraction is recall-limited** (NLP recall ≈0.81, AJCC edition not captured — `data-model-reference.md`),
  and the lymph-node fallback (criterion 3's "# nodes >0") is itself stand-alone and recall-limited
  (see (D)). Step 3 therefore attrits partly for *data-capture* reasons rather than biology — the binding
  constraint the request doesn't acknowledge. (In the cohort **columns**, this bites wherever EO is thinner.)
- **(F) Metastatic exclusion is ICD-coded (C77/78/79) — consistent for Table 1a, but under-detects.** This
  is a per-table point, not a blanket "RDQ ignores EO mets": **Table 1a** leverages EHR codes, so an
  ICD-only mets exclusion is internally consistent there; the concern is that ICD codes alone **cannot use
  the EO note-derived metastatic-location evidence**, so some metastatic patients may be wrongly **retained**
  in 1a. (Separately, EO metastasis extraction is itself recall-limited — Optum reports the **EO Metastasis
  *concept* NLP recall ≈0.60**, `data-model-reference.md` "NLP model performance"; note this is the
  EO-concept recall, **not** an ICD-code recall figure.) **Implementation (RDQ-1 path):** Table 1a step 4
  is **ICD-only** (`flag_mets_rdq1` and R `rdq1_step4_nomets` use structured C77/78/79 alone) — faithful
  to the `[from EHR]` tag; the **EO `onc_metastatic_location`** overlay is deliberately **excluded from 1a**
  and reserved for **Table 1b** / a curated-sensitivity run (§5.5). (The *generic* Case A notebook/lib still
  union the two — that's the generic demo, not the RDQ-1 1a column.)
- **(G) "dMMR+ or MSI-H" with no result-interpretation rule.** The assay semantics are non-trivial and a
  classic error source: MMR-IHC **"positive" = retained protein = proficient (must NOT count)**, whereas
  **loss/absent/deficient = dMMR**; MSI **high/unstable = MSI-H**. Leaving "positive" undefined invites
  miscounting in either direction. **In the repo:** both the notebook and the reference SQL apply this rule
  correctly — `lib_cancer_cohort.sql` applies the per-concept rule in both the EO-biomarker and
  structured-Lab arms, and `caseA_colon_attrition_databricks.py` excludes MMR "positive", so both paths agree.

### Undefined / unstated assumptions
- **(B) No ≥2-confirmed-diagnosis rule and no rule-out handling.** Single-code cancer identification is
  non-standard; RWE convention (and `methodology-and-derivation.md`) requires **≥2 confirmed dx on
  separate dates** to suppress rule-out/false-positive coding. As written, a single rule-out code qualifies.
- **(C) "Newly diagnosed" with no clean-baseline washout.** "≤183 d before surgery" bounds the window
  *forward* but sets no *backward* lookback to ensure the diagnosis is genuinely **incident** — prevalent or
  recurrent disease can enter. (The repo's `caseA` step 2 adds an explicit pre-window washout; the request
  omits it.)
- **(H) The IO timing window is effectively open-ended.** "After diagnosis and prior to metastatic
  diagnosis" — but criterion 4 already removes anyone with mets before surgery, and for never-metastatic
  patients there is **no upper bound**, so IO administered far downstream (or for another indication) still
  qualifies.
- **(L) Index date is not specified** in the request, yet every temporal criterion (newly-dx window, IO timing,
  mets-before-surgery) needs an anchor. *(→ §5.7.)*
- **(M) Cohort 4 "Market Clarity IHD" is a different denominator.** IHD is the broad de-identified
  individual-data universe — stacking it as a 4th column of the *same* attrition compares a different source
  population against the EHR+EO cohorts (apples-to-oranges within one table).
- **(N) Study period 2020–2025 on a 2007–2025 database — define the lookback rule.** *Conditional:* if
  **all** input criteria are restricted to 2020+, patients operated early in the window have **truncated
  lookback** (pre-surgery dx history, prior IO) — a left-truncation effect. If instead the study period
  anchors only the **surgery/index** date and **pre-2020 data may be used for baseline/history**, the effect
  is mitigated. The request doesn't state which — so **define explicitly whether pre-2020 data is in scope for
  lookback**. *(Implemented — the code applies the study window to the **surgery/index date only**
  (the mitigated reading): R `RDQ1_STUDY_START`/`RDQ1_STUDY_END` = 2020-01-01/2025-12-31 in step 1, and SQL
  `params_rdq1.study_start`/`study_end`. Togglable (set to `NULL`/widen to restrict-all or disable). Pre-2020
  lookback stays usable by default. Confirm the agreed lookback rule.)*
- **(K) C18 subcodes only — omitting C18.1 (appendix) and C19/C20 (rectosigmoid/rectum).** Defensible for "colon,"
  but the **C18.1 omission looks inadvertent** and the rectosigmoid boundary is an unstated scoping choice.
- **(O) "Closed" Optum claims and the 2a/2b split are undefined.** Cohort 2 says Optum Medical & Rx
  "(closed)" claims, and Table 1a's header shows "Cohort 2a" — neither the "closed" qualifier nor the a/b
  distinction is defined in the request.

> **Net:** items **A, D, J, F, G** can *change which patients are in the cohort* and deserve an explicit
> requestor/clinical decision before any run; **B, C, E, H, L–O** are definitional gaps to pin down so the
> numbers are interpretable. None of this blocks the *method* — but it does mean Table 1a/1b would encode
> these choices, so they must be agreed, not assumed.

## 5. Reconciliation: RDQ-1 (authoritative) vs. the current pipeline

RDQ-1 supplies explicit code lists and a cohort layout that **differ** from the generic Case A model in
the repo (which uses a looser colorectal definition). These must be reconciled **before** a run, and
a few need a **requestor/clinical decision** — they are not the analyst's to silently change.

> **Most of §5 is implemented** as flagged, togglable defaults (see §8) — the value sets, the C77 toggle,

### 5.1 Metastatic-exclusion codes — **C77 conflict (decision needed)**
- **RDQ-1 (criterion 4):** exclude **C77.x, C78.x, C79.x** prior to surgery.
- **Current pipeline:** distant-only **C78–C79**, and **deliberately keeps C77** (secondary lymph-node
  neoplasm) — see `lib_cancer_cohort.sql` `flag_mets` and `caseA_colon.sql` header.
- **Why it matters:** the cohort is **locally advanced, N+ (T1–4 N1–2)**. C77 ("secondary malignant
  neoplasm of lymph nodes") **can overlap with** the regional nodal disease the cohort is meant to retain,
  so excluding C77 wholesale risks dropping target N+ patients. RDQ-1 resolves the repo's open question —
  but in the **opposite** direction.
- **Action:** confirm with the requestor. If C77 is truly to be excluded, change `flag_mets` to
  `^C7[789]`. **Recommend flagging the clinical tension** (regional nodal disease ≠ distant metastasis)
  before doing so.

### 5.2 Colon diagnosis codes — **use the explicit enumerated list**
- **RDQ-1 diagnosis code list:** an **enumerated list** — C18, C18.0, C18.2, C18.3, C18.4, C18.5, C18.6,
  C18.7, C18.8, C18.9. Note **C18.1 (appendix) is *not* in the list**.
- **Current pipeline:** `^C18|^C19|^C20` (colon **+ rectosigmoid + rectum**); notebook adds C19/C20.
- **Action:** match the **exact enumerated diagnosis codes**, *not* a `^C18` prefix — a prefix
  regex would silently re-include **C18.1**, which the list omits. So either enumerate the listed
  subcodes or use a prefix-minus-C18.1 pattern. Confirm with the requestor whether the C18.1 omission (and
  the exclusion of rectosigmoid/rectum C19/C20) is intentional before finalizing.

### 5.3 IO agent list — **extend to all 11**
- **RDQ-1 IO code list:** atezolizumab (J9022), avelumab (J9023), cemiplimab (J9119),
  durvalumab (J9173), nivolumab (J9299 / J9289), pembrolizumab (J9271), ipilimumab (J9228),
  dostarlimab (J9272), toripalimab (J3263), tislelizumab (J9329), cosibelimab (J9275) — **plus NDC codes**.
- **Current pipeline:** `vs_drug` has 7 IO agents; the notebook's `IO_DRUGS` lists only 4 (pembro, nivo,
  dostar, ipi).
- **Action — bigger than adding value-set rows.** Today step 6 joins `flag_drug` → `vs_drug` **by generic
  name only** (`generic_desc`), so honouring the RDQ IO list means: (i) add the **4 missing agents**

"""
Extract rdq1_transfer.txt back into files.

Usage:
    python rdq1_extract.py [bundle.txt] [output_dir]

Defaults: bundle = rdq1_transfer.txt, output_dir = current directory.

UTF-8 text; each file's content sits between a '>>>>> FILE: <path>' marker line
and the next marker. This extractor:
  * reads tolerant of LF, CRLF or CR line endings (a Windows checkout that
    converted endings still extracts cleanly);
  * writes every file as UTF-8 with LF endings in binary mode (no platform
    translation, nothing doubled);
  * rejects absolute paths and '..' segments (writes only under output_dir).
Pure standard library, Python 3.x.
"""
import sys, os

MARK = ">>>>> FILE: "
END  = ">>>>> END"

def main():
    bundle  = sys.argv[1] if len(sys.argv) > 1 else "rdq1_transfer.txt"
    out_dir = sys.argv[2] if len(sys.argv) > 2 else "."
    if not os.path.isfile(bundle):
        print("ERROR: bundle not found:", bundle); sys.exit(1)
    with open(bundle, "r", encoding="utf-8") as fh:
        lines = fh.read().splitlines()          # handles \n, \r\n, \r
    path, buf, count = None, [], 0
    def flush():
        nonlocal count
        if path is None: return
        parts = [p for p in path.replace("\\", "/").split("/") if p not in ("", ".")]
        if os.path.isabs(path) or any(p == ".." for p in parts) or not parts:
            print("  SKIP unsafe path:", path); return
        target = os.path.join(out_dir, *parts)
        parent = os.path.dirname(target)
        if parent: os.makedirs(parent, exist_ok=True)
        content = "\n".join(buf) + "\n"          # LF endings + trailing newline
        with open(target, "wb") as out:          # binary -> no CRLF translation
            out.write(content.encode("utf-8"))
        print("  wrote", target, "(%d lines)" % len(buf))
        count += 1
    for line in lines:
        if line.startswith(MARK):
            flush(); path = line[len(MARK):].strip(); buf = []
        elif line.strip() == END:
            flush(); path = None
        elif path is not None:
            buf.append(line)
    flush()                                      # handles a bundle with no END marker
    print("done: %d files into %s" % (count, os.path.abspath(out_dir)))

if __name__ == "__main__":
    main()
