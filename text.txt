# FRalpha status at index by treatment (Mirvetuximab vs ICC), for ovarian patients on
# 2L-4L lines (ICC restricted to single-agent). Builds Table 1 (counts and within-arm %).
# Runs on local data frames with dplyr. Give it a DBI `con` (or set FRA_SETUP_R), then:
#   Rscript fralpha_by_treatment.R

library(dplyr); library(tidyr); library(stringr)

# ---- config ----
catalog         <- Sys.getenv("FRA_CATALOG", "hive_metastore")
schema          <- "clnprw_flatiron_pano_ovar_use"
drugepisode_tbl <- "t_drugepisode_2026m06"
biomarker_tbl   <- "t_enh_ovbiomarkers_2026m06"   # confirm this is the FOLR1 snapshot to use
biomarker_name  <- "FOLR1/FRalpha"
index_lines     <- c(2L, 3L, 4L)
bio_window_days <- 14L
pct_scale       <- Sys.getenv("FRA_PCT_SCALE", "auto")   # fraction | percent | auto
source_mode     <- Sys.getenv("FRA_SOURCE_MODE", "dbi")  # dbi | local
local_dir       <- Sys.getenv("FRA_LOCAL_DIR", "")

ARM_ICC   <- "ICC (Paclitaxel, PLD, or topotecan)"
FRA_ORDER <- c("FRa high", "FRa low", "FRa moderate", "Unknown/untested")   # table row order
FRA_RANK  <- c("FRa high", "FRa moderate", "FRa low", "Unknown/untested")   # best to worst

this_dir <- tryCatch(dirname(sub("^--file=", "", grep("^--file=", commandArgs(FALSE), value = TRUE))),
                     error = function(e) ".")
if (!length(this_dir) || this_dir == "") this_dir <- "."
out_dir <- Sys.getenv("FRA_OUT_DIR", file.path(this_dir, "outputs"))

# ---- load a table into a local data frame ----
load_tbl <- function(tbl, cols) {
  if (source_mode == "local") {
    stopifnot(nzchar(local_dir))
    pq <- file.path(local_dir, paste0(tbl, ".parquet"))
    if (file.exists(pq)) return(as_tibble(arrow::read_parquet(pq, col_select = all_of(cols))))
    cs <- file.path(local_dir, paste0(tbl, ".csv"))
    if (file.exists(cs)) return(as_tibble(readr::read_csv(cs, show_col_types = FALSE))[cols])
    stop(sprintf("no extract for %s in %s", tbl, local_dir))
  }
  fqtn <- paste(c(catalog, schema, tbl), collapse = ".")
  as_tibble(DBI::dbGetQuery(con, sprintf("SELECT %s FROM %s", paste(cols, collapse = ", "), fqtn)))
}

# ---- FRalpha result: percent, intensity, class ----
# PercentStaining is stored as text; it can be a fraction (0-1) or a percent (0-100).
# Pick the scale from the data, and stop if it is mixed rather than guess.
resolve_scale <- function(num) {
  if (pct_scale %in% c("fraction", "percent")) return(pct_scale)
  v <- num[!is.na(num)]
  if (!length(v)) return("percent")
  if (max(v) <= 1) return("fraction")
  if (min(v) > 1)  return("percent")
  stop("PercentStaining has values on both sides of 1 — set FRA_PCT_SCALE to 'fraction' or 'percent'.")
}
pct_numeric <- function(x) suppressWarnings(as.numeric(str_replace_all(as.character(x), "[^0-9.\\-]", "")))
to_pct <- function(x, scale) { num <- pct_numeric(x); if (scale == "fraction") num * 100 else num }
to_intensity <- function(x) {
  s <- str_to_lower(as.character(x))
  # values come as "3+", "strong", or a bare "3" (dict lists "3 / 3+ / Strong staining")
  case_when(str_detect(s, "3\\s*\\+|strong")             ~ 3L,
            str_detect(s, "2\\s*\\+|moderate")           ~ 2L,
            str_detect(s, "1\\s*\\+|weak")               ~ 1L,
            str_detect(s, "negative|no staining")        ~ 0L,
            str_detect(s, "(^|[^0-9])3([^0-9]|$)")       ~ 3L,
            str_detect(s, "(^|[^0-9])2([^0-9]|$)")       ~ 2L,
            str_detect(s, "(^|[^0-9])1([^0-9]|$)")       ~ 1L,
            str_detect(s, "(^|[^0-9])0([^0-9]|$)")       ~ 0L,
            TRUE ~ NA_integer_)
}
fra_class <- function(pct, intensity) {
  case_when(pct >= 75 & intensity %in% c(2L, 3L)            ~ "FRa high",
            pct >= 50 & pct < 75 & intensity %in% c(2L, 3L) ~ "FRa moderate",
            (pct < 50) | (intensity <= 1L)                  ~ "FRa low",
            TRUE                                            ~ "Unknown/untested")
}

# ---- treatment arm from drug name ----
arm_of <- function(drugname) {
  d <- str_to_lower(drugname)
  case_when(str_detect(d, "mirvetuximab")                      ~ "Mirvetuximab",
            str_detect(d, "topotecan")                         ~ ARM_ICC,
            str_detect(d, "doxil|lipodox|caelyx")              ~ ARM_ICC,  # pegylated liposomal doxorubicin
            str_detect(d, "doxorubicin") & str_detect(d, "liposom|pegylat") ~ ARM_ICC,
            str_detect(d, "paclitaxel")                        ~ ARM_ICC,
            TRUE                                               ~ NA_character_)
}
is_nab      <- function(d) { d2 <- str_squish(str_to_lower(d))
  str_detect(d2, "paclitaxel") & str_detect(d2, "nab|protein[- ]?bound|albumin|abraxane") }
is_nonmaint <- function(x) str_to_lower(str_trim(as.character(x))) %in% c("false","f","n","no","0")
is_advanced <- function(s) str_to_lower(str_trim(coalesce(as.character(s), ""))) == "advanced"

# ---- build the markdown table ----
table1_md <- function(df, title) {
  d <- df %>% mutate(arm = factor(arm, c("Mirvetuximab", ARM_ICC)),
                     fra = factor(fra, FRA_ORDER)) %>%
    count(arm, fra, .drop = FALSE, name = "n") %>%
    group_by(arm) %>% mutate(tot = sum(n), pct = if_else(tot > 0, 100 * n / tot, NA_real_)) %>% ungroup()
  cell <- function(a, f) { r <- d[d$arm == a & d$fra == f, ]
    if (is.na(r$pct)) sprintf("%d", r$n) else sprintf("%d (%.1f%%)", r$n, r$pct) }
  nm <- sum(d$n[d$arm == "Mirvetuximab"]); ni <- sum(d$n[d$arm == ARM_ICC])
  out <- c(sprintf("### %s", title),
           sprintf("| FOLR1/FRalpha at index (N, %%) | Mirvetuximab (N = %d) | %s (N = %d) |", nm, ARM_ICC, ni),
           "|---|---|---|")
  for (f in FRA_ORDER) out <- c(out, sprintf("| %s | %s | %s |", f, cell("Mirvetuximab", f), cell(ARM_ICC, f)))
  paste(out, collapse = "\n")
}

# ---- run ----
run_analysis <- function() {
  if (!pct_scale %in% c("fraction", "percent", "auto"))
    stop("FRA_PCT_SCALE must be 'fraction', 'percent', or 'auto'.")
  if (!source_mode %in% c("dbi", "local"))
    stop("FRA_SOURCE_MODE must be 'dbi' or 'local'.")
  if (source_mode == "dbi" && !exists("con")) {
    setup <- Sys.getenv("FRA_SETUP_R")
    if (nzchar(setup) && file.exists(setup)) source(setup, local = FALSE)
    if (!exists("con"))
      stop("need a DBI `con` — create one, set FRA_SETUP_R to a script that makes it, or use FRA_SOURCE_MODE=local.")
  }

  drugep <- load_tbl(drugepisode_tbl,
                     c("patientid","drugname","linename","linenumber","linesetting","ismaintenancetherapy","episodedate"))
  biomk  <- load_tbl(biomarker_tbl,
                     c("patientid","biomarkername","specimenreceiveddate","resultdate","percentstaining","stainingintensity"))

  drugep <- drugep %>% filter(!is.na(patientid))    # valid patient rows only
  biomk  <- biomk  %>% filter(!is.na(patientid))

  # ---- qualifying treatment episodes ----
  # Split each line's non-maintenance drugs into agents (composite names split, blanks
  # dropped, no date filter so a dateless partner still counts). A line is single-agent
  # if it has one agent; a line carrying both arms (mirvetuximab + a qualifying ICC drug;
  # nab-paclitaxel is not a qualifying ICC) is a cross-arm combination. Keyed by patient+line.
  line_comp <- drugep %>%
    filter(linenumber %in% index_lines, is_nonmaint(ismaintenancetherapy)) %>%
    mutate(agent = str_split(str_squish(str_to_lower(drugname)), "\\s*[,/+;&|]\\s*")) %>%
    tidyr::unnest(agent) %>%
    filter(!is.na(agent), agent != "") %>%
    mutate(agent_arm = if_else(is_nab(agent), NA_character_, arm_of(agent))) %>%
    group_by(patientid, linenumber) %>%
    summarise(n_agents = n_distinct(agent),
              n_arms   = n_distinct(agent_arm[!is.na(agent_arm)]), .groups = "drop")

  base_ep <- drugep %>% mutate(episodedate = as.Date(episodedate)) %>%
    filter(linenumber %in% index_lines, is_nonmaint(ismaintenancetherapy), !is.na(episodedate))
  cand    <- base_ep %>% mutate(arm = arm_of(drugname)) %>% filter(!is.na(arm))   # any target drug (incl. nab)
  cand_nn <- cand %>% filter(arm == "Mirvetuximab" | !is_nab(drugname))           # nab disqualifies ICC only
  in_line <- cand_nn %>% left_join(line_comp, by = c("patientid","linenumber"))
  no_xarm <- in_line %>% filter(n_arms <= 1)                                      # drop cross-arm (mirv + ICC) lines
  qual    <- no_xarm %>% filter(arm == "Mirvetuximab" | n_agents == 1)            # ICC must be single-agent
  n_nab_excl   <- n_distinct(cand$patientid)    - n_distinct(cand_nn$patientid)   # nab-only patients
  n_xarm_excl  <- n_distinct(cand_nn$patientid) - n_distinct(no_xarm$patientid)   # cross-arm-line-only patients
  n_combo_excl <- n_distinct(no_xarm$patientid) - n_distinct(qual$patientid)      # ICC-combination-only patients

  # ---- FRalpha per patient (score once; apply the window per cohort) ----
  folr1 <- biomk %>% filter(str_to_upper(str_trim(biomarkername)) == str_to_upper(biomarker_name))
  if (nrow(folr1) == 0)
    stop(sprintf("no records match biomarkername='%s' in %s — check the table and name.", biomarker_name, biomarker_tbl))
  scale    <- resolve_scale(pct_numeric(folr1$percentstaining))
  if (pct_scale == "auto" && scale == "fraction")
    message("PercentStaining read as fractions (all values <=1); set FRA_PCT_SCALE explicitly for the final run.")
  pct_norm <- to_pct(folr1$percentstaining, scale)
  if (any(pct_norm > 100 + 1e-9 | pct_norm < 0, na.rm = TRUE))
    stop(sprintf("PercentStaining out of 0-100 after scaling (%.2f..%.2f) — check FRA_PCT_SCALE.",
                 min(pct_norm, na.rm = TRUE), max(pct_norm, na.rm = TRUE)))
  fin     <- pct_norm[is.finite(pct_norm)]
  if (!length(fin)) message("no usable PercentStaining — high/moderate not possible; using intensity only.")
  pct_rng <- if (length(fin)) sprintf("%.3g..%.3g", min(fin), max(fin)) else "n/a"
  scored <- folr1 %>%
    mutate(test_date = pmax(as.Date(specimenreceiveddate), as.Date(resultdate), na.rm = TRUE),
           rec = fra_class(pct_norm, to_intensity(stainingintensity)))

  # highest FRalpha among a patient's records inside the window; none -> Unknown
  attach_fra <- function(pop) {
    f <- scored %>% inner_join(select(pop, patientid, index_date), by = "patientid") %>%
      filter(!is.na(test_date), test_date <= index_date + bio_window_days) %>%
      mutate(rank = match(rec, FRA_RANK)) %>%
      group_by(patientid) %>% summarise(rank = min(rank), .groups = "drop") %>%
      transmute(patientid, fra = FRA_RANK[rank])
    pop %>% left_join(f, by = "patientid") %>% mutate(fra = coalesce(fra, "Unknown/untested"))
  }

  # first qualifying dose sets arm and index; drop ties where the first date has both arms
  build_cohort <- function(q) {
    conflict <- q %>% group_by(patientid) %>% filter(episodedate == min(episodedate)) %>%
      summarise(n = n_distinct(arm), .groups = "drop") %>% filter(n > 1) %>% pull(patientid)
    pop <- q %>% filter(!patientid %in% conflict) %>%
      arrange(patientid, episodedate, drugname) %>% group_by(patientid) %>% slice(1L) %>% ungroup() %>%
      transmute(patientid, arm, index_date = episodedate, linesetting)
    list(pop = attach_fra(pop), n_conflict = length(conflict))
  }

  all_c <- build_cohort(qual)                                     # any setting
  adv_c <- build_cohort(qual %>% filter(is_advanced(linesetting)))# advanced only, chosen independently

  t_all <- table1_md(all_c$pop, "Table 1a — all 2L-4L lines (any setting)")
  t_adv <- table1_md(adv_c$pop, "Table 1b — advanced setting only (first qualifying advanced dose)")
  summary <- paste(
    sprintf("Cohort 1a: %d patients (Mirv %d, ICC %d); same-day conflicts excluded %d.",
            nrow(all_c$pop), sum(all_c$pop$arm == "Mirvetuximab"), sum(all_c$pop$arm == ARM_ICC), all_c$n_conflict),
    sprintf("Cohort 1b (advanced): %d patients (Mirv %d, ICC %d); same-day conflicts excluded %d.",
            nrow(adv_c$pop), sum(adv_c$pop$arm == "Mirvetuximab"), sum(adv_c$pop$arm == ARM_ICC), adv_c$n_conflict),
    sprintf("Excluded: %d nab-only, %d cross-arm-line-only, %d ICC-combination-only patients.",
            n_nab_excl, n_xarm_excl, n_combo_excl),
    sprintf("FOLR1 records: %d rows, %d patients. PercentStaining scale=%s (range %s).",
            nrow(folr1), n_distinct(folr1$patientid), scale, pct_rng),
    sprintf("Source: %s / %s.%s.", drugepisode_tbl, biomarker_tbl, biomarker_name),
    sep = "\n")

  cat("\n", summary, "\n\n", t_all, "\n\n", t_adv, "\n", sep = "")
  dir.create(out_dir, showWarnings = FALSE, recursive = TRUE)
  writeLines(c(summary, "", t_all, "", t_adv), file.path(out_dir, "fralpha_by_treatment.md"))
  message(sprintf("wrote %s", file.path(out_dir, "fralpha_by_treatment.md")))
  invisible(list(all = all_c$pop, adv = adv_c$pop, table1_all = t_all, table1_adv = t_adv))
}

if (!nzchar(Sys.getenv("FRA_DEFINE_ONLY"))) run_analysis()

-- FRalpha analysis — data profiling + QC (Databricks / Spark SQL).
-- Run these first to check the data; section 10 rebuilds Table 1 as an independent cross-check.
-- Change the catalog below if yours is not hive_metastore.
USE CATALOG hive_metastore;
USE SCHEMA  clnprw_flatiron_pano_ovar_use;

-- =====================================================================
-- 0. Which FOLR1 biomarker snapshots exist? Pick the one to use.
SHOW TABLES LIKE 't_enh_ovbiomarkers_*';
SHOW TABLES LIKE 't_drugepisode_*';

-- =====================================================================
-- 1. biomarkername values — confirm the exact string and find any variants/whitespace.
SELECT biomarkername, count(*) AS n, count(DISTINCT patientid) AS pts
FROM t_enh_ovbiomarkers_2026m06
GROUP BY biomarkername
ORDER BY n DESC;

-- 2. FOLR1 record + patient counts (the zero-match guard depends on this being > 0).
SELECT count(*) AS folr1_rows, count(DISTINCT patientid) AS folr1_pts
FROM t_enh_ovbiomarkers_2026m06
WHERE upper(trim(biomarkername)) = 'FOLR1/FRALPHA';

-- =====================================================================
-- 3. PercentStaining scale + quality (fraction vs percent).
--    Look at: how many parse; how many <=1 vs >1 (mixed => ambiguous, set FRA_PCT_SCALE
--    explicitly); negatives (should be 0); min/max.
WITH f AS (
  SELECT try_cast(regexp_replace(percentstaining, '[^0-9.-]', '') AS DOUBLE) AS p, percentstaining
  FROM t_enh_ovbiomarkers_2026m06
  WHERE upper(trim(biomarkername)) = 'FOLR1/FRALPHA'
)
SELECT count(*)                                   AS folr1_rows,
       count(p)                                   AS parseable,
       sum(CASE WHEN p IS NULL THEN 1 ELSE 0 END) AS unparseable_or_null,
       sum(CASE WHEN p <= 1  THEN 1 ELSE 0 END)   AS le_1,          -- looks fractional
       sum(CASE WHEN p >  1  THEN 1 ELSE 0 END)   AS gt_1,          -- looks like a percent
       sum(CASE WHEN p <  0  THEN 1 ELSE 0 END)   AS negatives,     -- expect 0
       min(p) AS min_p, max(p) AS max_p
FROM f;
-- distinct raw values, to eyeball formatting ("0.75" vs "75" vs "75%").
SELECT percentstaining, count(*) AS n
FROM t_enh_ovbiomarkers_2026m06
WHERE upper(trim(biomarkername)) = 'FOLR1/FRALPHA'
GROUP BY percentstaining ORDER BY n DESC LIMIT 50;

-- 4. StainingIntensity values — confirm 0 / 1+ / 2+ / 3+ coding; catch surprises.
SELECT stainingintensity, count(*) AS n
FROM t_enh_ovbiomarkers_2026m06
WHERE upper(trim(biomarkername)) = 'FOLR1/FRALPHA'
GROUP BY stainingintensity ORDER BY n DESC;

-- =====================================================================
-- 5. Drug codelist — confirm the exact drugname strings the arm matcher relies on,
--    esp. PLD (must be PEGYLATED) and paclitaxel vs nab-paclitaxel.
SELECT drugname, count(*) AS episodes, count(DISTINCT patientid) AS pts,
       CASE
         WHEN lower(drugname) LIKE '%mirvetuximab%' THEN 'Mirvetuximab'
         WHEN lower(drugname) LIKE '%topotecan%'    THEN 'ICC:topotecan'
         WHEN lower(drugname) LIKE '%doxil%' OR lower(drugname) LIKE '%lipodox%'
              OR lower(drugname) LIKE '%caelyx%'    THEN 'ICC:PLD-brand'
         WHEN lower(drugname) LIKE '%doxorubicin%' AND
              (lower(drugname) LIKE '%liposom%' OR lower(drugname) LIKE '%pegylat%') THEN 'ICC:PLD-generic'
         WHEN lower(drugname) LIKE '%paclitaxel%' AND
              (lower(drugname) LIKE '%nab%' OR lower(drugname) LIKE '%protein%bound%'
               OR lower(drugname) LIKE '%albumin%' OR lower(drugname) LIKE '%abraxane%') THEN 'nab-paclitaxel (excluded)'
         WHEN lower(drugname) LIKE '%paclitaxel%' THEN 'ICC:paclitaxel'
         ELSE 'other' END AS maps_to
FROM t_drugepisode_2026m06
WHERE lower(drugname) LIKE '%mirvetuximab%' OR lower(drugname) LIKE '%paclitaxel%'
   OR lower(drugname) LIKE '%doxorubicin%'  OR lower(drugname) LIKE '%topotecan%'
   OR lower(drugname) LIKE '%doxil%' OR lower(drugname) LIKE '%lipodox%' OR lower(drugname) LIKE '%caelyx%'
GROUP BY drugname ORDER BY maps_to, episodes DESC;
-- 5b. EVERY distinct drugname on 2L-4L non-maintenance lines (no pattern filter) — so an
--     unknown brand-only name for a target drug is visible, not hidden by the matcher.
SELECT drugname, count(*) AS episodes, count(DISTINCT patientid) AS pts
FROM t_drugepisode_2026m06
WHERE linenumber IN (2,3,4) AND lower(trim(ismaintenancetherapy)) IN ('false','f','n','no','0')
GROUP BY drugname ORDER BY episodes DESC;

-- 6. ismaintenancetherapy values — confirm True/False coding; count NULL/blank
--    (these do NOT count as non-maintenance).
SELECT ismaintenancetherapy, count(*) AS n FROM t_drugepisode_2026m06
GROUP BY ismaintenancetherapy ORDER BY n DESC;

-- 7. linesetting values — confirm the exact "Advanced" label used for Table 1b.
SELECT linesetting, count(*) AS n, count(DISTINCT patientid) AS pts
FROM t_drugepisode_2026m06
GROUP BY linesetting ORDER BY n DESC;

-- =====================================================================
-- 8. Line composition / single-agent rule.
--    (a) distribution of distinct drugs per line instance (patient,line,setting).
WITH la AS (   -- split composite names into agents; drop blanks
  SELECT patientid, linenumber, count(DISTINCT agent) AS n_agents
  FROM (SELECT patientid, linenumber,
               explode(split(regexp_replace(lower(trim(drugname)), '\\s+', ' '), '\\s*[,/+;&|]\\s*')) AS agent
        FROM t_drugepisode_2026m06
        WHERE linenumber IN (2,3,4)
          AND lower(trim(ismaintenancetherapy)) IN ('false','f','n','no','0')
          AND drugname IS NOT NULL AND trim(drugname) <> '')
  WHERE agent <> ''
  GROUP BY patientid, linenumber
)
SELECT n_agents, count(*) AS line_instances FROM la GROUP BY n_agents ORDER BY n_agents;
-- (b) does a (patient,line) span more than one linesetting? Composition is keyed by
--     patient+line, so >0 rows means a line's setting is ambiguous for cohort assignment.
SELECT count(*) AS patient_lines_spanning_settings FROM (
  SELECT patientid, linenumber FROM t_drugepisode_2026m06
  WHERE linenumber IN (2,3,4)
  GROUP BY patientid, linenumber HAVING count(DISTINCT coalesce(linesetting,'')) > 1
);

-- =====================================================================
-- 9. Cohort funnel — patient counts at each filter step. Reconcile against the R
--    run summary (Cohort 1a N, nab-only, combination-only).
WITH base AS (
  SELECT * FROM t_drugepisode_2026m06
  WHERE linenumber IN (2,3,4)
    AND lower(trim(ismaintenancetherapy)) IN ('false','f','n','no','0')
    AND episodedate IS NOT NULL
), tagged AS (
  SELECT b.*,
    CASE
      WHEN lower(drugname) LIKE '%mirvetuximab%' THEN 'Mirvetuximab'
      WHEN lower(drugname) LIKE '%topotecan%' THEN 'ICC'
      WHEN lower(drugname) LIKE '%doxil%' OR lower(drugname) LIKE '%lipodox%' OR lower(drugname) LIKE '%caelyx%' THEN 'ICC'
      WHEN lower(drugname) LIKE '%doxorubicin%' AND (lower(drugname) LIKE '%liposom%' OR lower(drugname) LIKE '%pegylat%') THEN 'ICC'
      WHEN lower(drugname) LIKE '%paclitaxel%' THEN 'ICC' ELSE NULL END AS arm,
    (lower(drugname) LIKE '%paclitaxel%' AND (lower(drugname) LIKE '%nab%' OR lower(drugname) LIKE '%protein%bound%'
       OR lower(drugname) LIKE '%albumin%' OR lower(drugname) LIKE '%abraxane%')) AS is_nab
  FROM base b
), la AS (
  SELECT patientid, linenumber,
         count(DISTINCT agent) AS n_agents,
         count(DISTINCT CASE
           WHEN agent LIKE '%mirvetuximab%' THEN 'M'
           WHEN agent LIKE '%paclitaxel%' AND (agent LIKE '%nab%' OR agent LIKE '%protein%bound%'
                OR agent LIKE '%albumin%' OR agent LIKE '%abraxane%') THEN NULL
           WHEN agent LIKE '%topotecan%' OR agent LIKE '%doxil%' OR agent LIKE '%lipodox%' OR agent LIKE '%caelyx%'
             OR (agent LIKE '%doxorubicin%' AND (agent LIKE '%liposom%' OR agent LIKE '%pegylat%'))
             OR agent LIKE '%paclitaxel%' THEN 'I' END) AS n_arms
  FROM (SELECT patientid, linenumber,
               explode(split(regexp_replace(lower(trim(drugname)), '\\s+', ' '), '\\s*[,/+;&|]\\s*')) AS agent
        FROM t_drugepisode_2026m06
        WHERE linenumber IN (2,3,4) AND lower(trim(ismaintenancetherapy)) IN ('false','f','n','no','0')
          AND drugname IS NOT NULL AND trim(drugname) <> '')
  WHERE agent <> ''
  GROUP BY patientid, linenumber
)
SELECT
  count(DISTINCT CASE WHEN arm IS NOT NULL THEN patientid END)                                  AS with_target_drug,
  count(DISTINCT CASE WHEN arm IS NOT NULL AND (arm='Mirvetuximab' OR NOT is_nab) THEN patientid END) AS after_nab_drop,
  count(DISTINCT CASE WHEN arm IS NOT NULL AND (arm='Mirvetuximab' OR NOT is_nab) AND la.n_arms <= 1
                       AND (arm='Mirvetuximab' OR la.n_agents = 1) THEN patientid END)          AS single_agent_qualified
FROM tagged t LEFT JOIN la ON t.patientid=la.patientid AND t.linenumber=la.linenumber;

-- =====================================================================
-- 10. Independent Table 1 in SQL — an alternate build for cross-checking.
--     Set the scale in `params` to match Section 3 ('fraction' or 'percent').
--     Produces BOTH cohorts: cohort='all' (Table 1a) and 'adv' (Table 1b).
WITH params AS (SELECT 'percent' AS pct_scale),   -- <-- set from Section 3 before trusting cells
la AS (
  SELECT patientid, linenumber,
         count(DISTINCT agent) AS n_agents,
         count(DISTINCT CASE
           WHEN agent LIKE '%mirvetuximab%' THEN 'M'
           WHEN agent LIKE '%paclitaxel%' AND (agent LIKE '%nab%' OR agent LIKE '%protein%bound%'
                OR agent LIKE '%albumin%' OR agent LIKE '%abraxane%') THEN NULL   -- nab: not a qualifying ICC arm
           WHEN agent LIKE '%topotecan%' OR agent LIKE '%doxil%' OR agent LIKE '%lipodox%' OR agent LIKE '%caelyx%'
             OR (agent LIKE '%doxorubicin%' AND (agent LIKE '%liposom%' OR agent LIKE '%pegylat%'))
             OR agent LIKE '%paclitaxel%' THEN 'I' END) AS n_arms
  FROM (SELECT patientid, linenumber,
               explode(split(regexp_replace(lower(trim(drugname)), '\\s+', ' '), '\\s*[,/+;&|]\\s*')) AS agent
        FROM t_drugepisode_2026m06
        WHERE linenumber IN (2,3,4) AND lower(trim(ismaintenancetherapy)) IN ('false','f','n','no','0')
          AND drugname IS NOT NULL AND trim(drugname) <> '')
  WHERE agent <> ''
  GROUP BY patientid, linenumber
),
qualifying AS (
  SELECT d.patientid, d.episodedate, d.drugname,
         regexp_replace(lower(trim(coalesce(d.linesetting,''))), '\\s+', ' ') AS linesetting,
    CASE
      WHEN lower(d.drugname) LIKE '%mirvetuximab%' THEN 'Mirvetuximab'
      ELSE 'ICC (Paclitaxel, PLD, or topotecan)' END AS arm
  FROM t_drugepisode_2026m06 d JOIN la
    ON d.patientid=la.patientid AND d.linenumber=la.linenumber
  WHERE d.linenumber IN (2,3,4)
    AND lower(trim(d.ismaintenancetherapy)) IN ('false','f','n','no','0')
    AND d.episodedate IS NOT NULL
    AND la.n_arms <= 1                                                -- drop cross-arm (mirv + ICC) lines
    AND (lower(d.drugname) LIKE '%mirvetuximab%' OR la.n_agents = 1)  -- ICC single-agent; mirvetuximab may be combined
    AND ( lower(d.drugname) LIKE '%mirvetuximab%' OR lower(d.drugname) LIKE '%topotecan%'
       OR lower(d.drugname) LIKE '%doxil%' OR lower(d.drugname) LIKE '%lipodox%' OR lower(d.drugname) LIKE '%caelyx%'
       OR (lower(d.drugname) LIKE '%doxorubicin%' AND (lower(d.drugname) LIKE '%liposom%' OR lower(d.drugname) LIKE '%pegylat%'))
       OR lower(d.drugname) LIKE '%paclitaxel%' )
    AND (lower(d.drugname) LIKE '%mirvetuximab%'                      -- nab disqualifies ICC only
         OR NOT (lower(d.drugname) LIKE '%paclitaxel%' AND (lower(d.drugname) LIKE '%nab%' OR lower(d.drugname) LIKE '%protein%bound%'
                 OR lower(d.drugname) LIKE '%albumin%' OR lower(d.drugname) LIKE '%abraxane%')))
),
qual_c AS (   -- both cohorts: 'all' (any setting), 'adv' (advanced only)
  SELECT 'all' AS cohort, q.* FROM qualifying q
  UNION ALL
  SELECT 'adv' AS cohort, q.* FROM qualifying q WHERE lower(trim(q.linesetting)) = 'advanced'
),
first_dt AS (SELECT cohort, patientid, min(episodedate) AS idx FROM qual_c GROUP BY cohort, patientid),
conflict AS (   -- earliest qualifying date carries both arms -> drop the patient
  SELECT q.cohort, q.patientid FROM qual_c q JOIN first_dt f
    ON q.cohort=f.cohort AND q.patientid=f.patientid AND q.episodedate=f.idx
  GROUP BY q.cohort, q.patientid HAVING count(DISTINCT q.arm) > 1
),
picked AS (   -- first qualifying dose per (cohort, patient), conflicts removed
  SELECT cohort, patientid, arm, index_date FROM (
    SELECT q.cohort, q.patientid, q.arm, q.episodedate AS index_date,
           row_number() OVER (PARTITION BY q.cohort, q.patientid ORDER BY q.episodedate, q.drugname) AS rn
    FROM qual_c q LEFT ANTI JOIN conflict c ON q.cohort=c.cohort AND q.patientid=c.patientid
  ) WHERE rn = 1
),
scored AS (   -- FOLR1 record class + biomarker date
  SELECT b.patientid,
    greatest(b.specimenreceiveddate, b.resultdate) AS test_date,
    CASE
      WHEN pct >= 75 AND inten IN (2,3) THEN 1                 -- rank 1 = high
      WHEN pct >= 50 AND pct < 75 AND inten IN (2,3) THEN 2    -- moderate
      WHEN pct < 50 OR inten <= 1 THEN 3                       -- low
      ELSE 4 END AS rk
  FROM (
    SELECT patientid, specimenreceiveddate, resultdate,
      (CASE WHEN (SELECT pct_scale FROM params)='fraction'
            THEN try_cast(regexp_replace(percentstaining,'[^0-9.-]','') AS DOUBLE)*100
            ELSE try_cast(regexp_replace(percentstaining,'[^0-9.-]','') AS DOUBLE) END) AS pct,
      (CASE WHEN lower(stainingintensity) RLIKE '3\\s*\\+|strong'   THEN 3
            WHEN lower(stainingintensity) RLIKE '2\\s*\\+|moderate' THEN 2
            WHEN lower(stainingintensity) RLIKE '1\\s*\\+|weak'     THEN 1
            WHEN lower(stainingintensity) LIKE '%negative%' OR lower(stainingintensity) LIKE '%no staining%' THEN 0
            WHEN lower(stainingintensity) RLIKE '(^|[^0-9])3([^0-9]|$)' THEN 3
            WHEN lower(stainingintensity) RLIKE '(^|[^0-9])2([^0-9]|$)' THEN 2
            WHEN lower(stainingintensity) RLIKE '(^|[^0-9])1([^0-9]|$)' THEN 1
            WHEN lower(stainingintensity) RLIKE '(^|[^0-9])0([^0-9]|$)' THEN 0
            ELSE NULL END) AS inten
    FROM t_enh_ovbiomarkers_2026m06 WHERE upper(trim(biomarkername))='FOLR1/FRALPHA'
  ) b
),
fra AS (   -- highest class within window, per (cohort, patient)
  SELECT p.cohort, p.patientid,
         CASE min(s.rk) WHEN 1 THEN 'FRa high' WHEN 2 THEN 'FRa moderate' WHEN 3 THEN 'FRa low' ELSE 'Unknown/untested' END AS fra
  FROM picked p JOIN scored s ON p.patientid=s.patientid
  WHERE s.test_date IS NOT NULL AND s.test_date <= date_add(p.index_date, 14)
  GROUP BY p.cohort, p.patientid
)
SELECT pk.cohort, pk.arm,
       coalesce(fra.fra, 'Unknown/untested') AS fra_status,
       count(*) AS n
FROM picked pk LEFT JOIN fra ON pk.cohort=fra.cohort AND pk.patientid=fra.patientid
GROUP BY pk.cohort, pk.arm, coalesce(fra.fra, 'Unknown/untested')
ORDER BY pk.cohort, pk.arm, fra_status;
