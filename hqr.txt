#!/usr/bin/env python3
# ---------------------------------------------------------------------------
# unpack_bundle.py - reconstruct the pipeline folder from a readable bundle.
#
# Usage (Windows / VSCode terminal, PowerShell or cmd):
#     python unpack_bundle.py <bundle.txt> [output_dir]
#
#   <bundle.txt>   the readable text bundle (files between @@@@@ markers).
#   [output_dir]   where to write the files. Default: mm_lot_pipeline
#
# Notes:
#   * File contents in the bundle are verbatim UTF-8 (not encoded).
#   * Files are written with UTF-8 and LF ("\n") line endings, regardless of
#     what line endings the bundle picked up in transit -- so CRLF/LF is
#     normalised cleanly (R and VSCode on Windows are happy with LF).
#   * Standard library only. Python 3.6+.
# ---------------------------------------------------------------------------
import os
import re
import sys

FILE_MARK = "@@@@@ FILE >>> "
END_MARK = "@@@@@ END FILE"

# path line, then minimal body, then the END marker line (each on its own line)
BLOCK = re.compile(
    r"^" + re.escape(FILE_MARK) + r"(?P<path>[^\r\n]*)\r?\n"
    r"(?P<body>.*?)"
    r"\r?\n" + re.escape(END_MARK) + r"[ \t]*$",
    re.DOTALL | re.MULTILINE,
)


def die(msg):
    sys.stderr.write("unpack_bundle: " + msg + "\n")
    sys.exit(1)


def safe_join(root, rel):
    rel = rel.strip().replace("\\", "/")
    if not rel or rel.startswith("/") or ".." in rel.split("/"):
        die("unsafe path in bundle: " + repr(rel))
    return os.path.join(root, *rel.split("/"))


def main():
    if len(sys.argv) < 2:
        die("usage: python unpack_bundle.py <bundle.txt> [output_dir]")
    bundle = sys.argv[1]
    out_dir = sys.argv[2] if len(sys.argv) > 2 else "mm_lot_pipeline"
    if not os.path.isfile(bundle):
        die("bundle not found: " + bundle)

    with open(bundle, "r", encoding="utf-8") as fh:
        text = fh.read()

    written = 0
    for m in BLOCK.finditer(text):
        rel = m.group("path").strip()
        body = m.group("body")
        # Normalise any line endings the bundle may have picked up back to LF.
        body = body.replace("\r\n", "\n").replace("\r", "\n")
        dest = safe_join(out_dir, rel)
        parent = os.path.dirname(dest)
        if parent and not os.path.isdir(parent):
            os.makedirs(parent, exist_ok=True)
        # newline="" so the LF bytes in `body` are written exactly (no OS munging)
        with open(dest, "w", encoding="utf-8", newline="") as out:
            out.write(body)
            if not body.endswith("\n"):
                out.write("\n")
        written += 1

    if written == 0:
        die("no file blocks found (is this the right bundle?)")
    print("unpacked %d files into: %s" % (written, os.path.abspath(out_dir)))


if __name__ == "__main__":
    main()
